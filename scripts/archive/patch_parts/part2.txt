 * POST /api/subscription/cancel — programar la cancel·lació a FINAL DEL PERÍODE
 * (l'usuari manté l'accés fins a currentPeriodEnd, igual que al Portal de
 * Stripe). Equival a l'Edge Function `cancel-subscription` del encàrrec.
 *
 * Seguretat: l'usuari ve de requireVerifiedUser() (sessió server-side); només

  let user;
  try {
    user = await requireVerifiedUser();
  } catch (e) {
    if (isEmailNotVerifiedError(e)) {
      return NextResponse.json({ error: "EMAIL_NOT_VERIFIED" }, { status: 403 });
    }
    return NextResponse.json({ error: "UNAUTHENTICATED" }, { status: 401 });
  }
  const rl = await rateLimit(`sub-cancel:${user.id}`, 5, 60_000);
  if (!rl.ok) return NextResponse.json({ error: "TOO_MANY" }, { status: 429 });

  const sub = await db.subscription.findFirst({
    where: { userId: user.id, status: { in: ["active", "trialing"] } },
  });

    return NextResponse.json(
      { error: "NO_ACTIVE_SUBSCRIPTION", message: "No tens cap subscripció vigent." },
      { status: 404 }
    );
  }

  // DEMO o sessió sense sub de Stripe → només estat local (sense cobrament).
  const isRealStripe =
    !sub.demo &&
    Boolean(process.env.STRIPE_SECRET_KEY) &&
    Boolean(sub.stripeSubscriptionId) &&
    !String(sub.stripeSubscriptionId).startsWith("session_");

    if (isRealStripe) {

      const s = await stripe.subscriptions.update(sub.stripeSubscriptionId!, {
        cancel_at_period_end: true,
      });
      await db.subscription.update({
        where: { id: sub.id },
        data: { ...subscriptionLocalData(s), plan: sub.plan },
      });
    } else {
      await db.subscription.update({
        where: { id: sub.id },
        data: { cancelAtPeriodEnd: true },

  const state = await getSubscriptionState(user);
  return NextResponse.json({ ok: true, state });
}

  const ip = clientIp(req);
  const rl = await rateLimit(`lookup:${ip}`, 60, 60_000);
  if (!rl.ok) return NextResponse.json({ error: "TOO_MANY" }, { status: 429 });

  const body = await req.json().catch(() => null);
  const lat = parseFloat(body?.lat);
  const lng = parseFloat(body?.lng);
  const buildingId = body?.buildingId ? String(body.buildingId).slice(0, 40) : null;
  if (!isFinite(lat) || !isFinite(lng) || Math.abs(lat) > 90 || Math.abs(lng) > 180) {
    return NextResponse.json({ error: "BAD_POINT" }, { status: 400 });
  }

  const session = await getAuthUser();

  // 1) edifici (opcional)
  let building: Building | null = null;
  if (buildingId) {
    building = await db.building.findUnique({ where: { id: buildingId } });
  }

  // 2) municipi: per l'edifici o per consulta ICGC amb cau
  let munCode = building?.municipalityCode || null;
  if (!munCode) {
    munCode = await resolveMunicipalityCode(lat, lng);
  }
  if (!munCode) {
    return NextResponse.json({ error: "MUNICIPALITY_NOT_FOUND" }, { status: 404 });

  const mun = await db.municipality.findUnique({ where: { code: munCode } });

  if (!session) {
    await db.event

      .catch(() => {});

      municipality: { name: mun.name, comarca: mun.comarca, isTense: mun.isTense },
      teaser: {
        hasRent: !!mun.rentLatest,
        period: mun.rentPeriod,
        message: "REGISTRA'T_PER_VEURE",
      },
    });
  }

  // 3.5) verificació de correu obligatòria per a les funcions principals
  //      (Fase 1: registre i autenticació). El compte es pot obrir i explorar
  //      el mapa, però les consultes detallades exigeixen correu verificat.

        error: "QUOTA_EXCEEDED",
        quota: { used: quota.used, limit: quota.limit },

      },
      { status: 402 }

    kind: "full",
    building: building

          id: building.id,
          tipus: building.tipus,

          nom: building.nom,
          refcat,
          yearBuilt,
          avgRentEurMonth: building.avgRentEurMonth,

      isTense: mun.isTense,
      tenseList: mun.tenseList,
    },
    quota: { used: quota.used + 1, limit: quota.limit },
  });
}

/**
 * PATCH /api/account  { name?, locale? }        — edita el perfil (Fase 6)
 * DELETE /api/account { confirm: "ESBORRA" }    — eliminació del compte (RGPD)
 *
 * L'eliminació esborra dades personals (nom, correu, hash) i anonimitza els
 * registres d'ús que tenen valor estadístic (SearchLog/Event), preservant
 * agregats sense dades personals.
 */
export async function PATCH(req: Request) {
  let user;
  try {
    user = await requireUser();
  } catch {
    return NextResponse.json({ error: "UNAUTHENTICATED" }, { status: 401 });
  }
  const body = await req.json().catch(() => null);
  const data: { name?: string | null; locale?: string } = {};
  if (body?.name !== undefined) data.name = body.name === null ? null : sanitizeName(String(body.name));
  if (body?.locale === "es" || body?.locale === "ca") data.locale = body.locale;
  const updated = await db.user.update({ where: { id: user.id }, data });
  return NextResponse.json({
    ok: true,
    user: { id: updated.id, email: updated.email, name: updated.name, locale: updated.locale },
  });
}

export async function DELETE(req: Request) {
  let user;
  try {
    user = await requireUser();
  } catch {
    return NextResponse.json({ error: "UNAUTHENTICATED" }, { status: 401 });
  }
  const body = await req.json().catch(() => null);
  if (body?.confirm !== "ESBORRA") {
    return NextResponse.json({ error: "CONFIRM_REQUIRED" }, { status: 400 });
  }

  await db.$transaction([
    // dades personals directes
    db.favorite.deleteMany({ where: { userId: user.id } }),
    db.savedComparison.deleteMany({ where: { userId: user.id } }),
    db.passwordToken.deleteMany({ where: { userId: user.id } }),
    db.emailToken.deleteMany({ where: { userId: user.id } }),
    db.subscription.deleteMany({ where: { userId: user.id } }),
    db.subscriptionUsage.deleteMany({ where: { userId: user.id } }), // v22
    db.priceReport.deleteMany({ where: { userId: user.id } }),
    // pseudonimització d'ús històric (mètriques agregades sense PII)
    db.searchLog.updateMany({ where: { userId: user.id }, data: { userId: null } }),

    db.emailLog.deleteMany({ where: { to: user.email } }),
  ]);
  await revokeSessions(user.id);
  await db.user.delete({ where: { id: user.id } });
  await audit(null, "ACCOUNT_DELETED", user.id, { email: "(esborrat)" });
  return NextResponse.json({ ok: true });
}

export async function POST(req: Request) {

    // P1.3: MAI es cobra a un correu sense verificar (ni DEMO ni Stripe real)

      });

      {
        error: "STRIPE_NOT_CONFIGURED",
        message:
          "Stripe no está configurado todavía. El propietario debe conectar su cuenta de Stripe y definir STRIPE_SECRET_KEY / STRIPE_PRICE_* en .env.",
      },
      { status: 503 }
    );
  }

  // Stripe real (requiere precios creados en el panel de Stripe)
  // v22: noms canònics del encàrrec (STRIPE_PREMIUM_PRICE_ID /
  // STRIPE_PRO_AGENTS_PRICE_ID) amb retrocompatibilitat dels antics.

    process.env[plan.stripePriceEnv!] ||
    process.env[planId === "PRO" ? "STRIPE_PRICE_PRO" : "STRIPE_PRICE_PREMIUM"];
  if (!priceId) {
    return NextResponse.json(
      {
        error: "PRICE_NOT_CONFIGURED",
        message: `Falta ${plan.stripePriceEnv} en .env (créalo en el panel de Stripe como precio recurrente).`,
      },
      { status: 503 }
    );

    mode: "subscription",
    line_items: [{ price: priceId, quantity: 1 }],

    // retorn només consulta l'estat real de la sessió (Fase 7)

    cancel_url: `${origin}/#/subscripcio/cancelat`,
    metadata: { userId: user.id, plan: planId },
  });
  await db.event.create({ data: { type: "CHECKOUT_START", userId: user.id, meta: JSON.stringify({ plan: planId }) } }).catch(() => {});
  return NextResponse.json({ ok: true, mode: "STRIPE", url: cs.url });
}

 * Gestiona (Fase 7):
 *  - checkout.session.completed → activa la subscripció (mai des de la URL de retorn)
 *  - customer.subscription.updated/deleted → sincronitza estat i canvi de pla
 *  - invoice.paid → renovacions (actualitza període)
 *  - invoice.payment_failed → marca past_due
 *
 * IDEMPOTÈNCIA (P1.1): la fila StripeEvent fa de LOCK. S'insereix ABANS
 * d'aplicar efectes (claim atòmic per la clau única eventId) i S'ELIMINA si
 * el processament falla → Stripe reentrega i l'event torna a executar-se.
 * Abans, la fila es registrava sense rollback: un error posterior deixava
 * l'event «devorat» (reentregues rebutjades com a duplicat) i la subscripció
 * podia quedar dessincronitzada per sempre.
 */
export async function POST(req: Request) {
  const secret = process.env.STRIPE_SECRET_KEY;
  const whSecret = process.env.STRIPE_WEBHOOK_SECRET;
  if (!secret) return NextResponse.json({ error: "STRIPE_NOT_CONFIGURED" }, { status: 503 });

  const stripe = new Stripe(secret);
  const sig = req.headers.get("stripe-signature");
  const raw = await req.text();
  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(raw, sig || "", whSecret || "");
  } catch {
    return NextResponse.json({ error: "BAD_SIGNATURE" }, { status: 400 });
  }

  // --- idempotència P1.1: claim → processar → alliberar si falla ---
  const claimed = await db.stripeEvent
    .create({ data: { eventId: event.id, type: event.type } })
    .then(() => true)
    .catch(() => false);
  if (!claimed) {
    // ja processat amb ÈXIT (reentrega): resposta sense efectes
    return NextResponse.json({ received: true, duplicate: true });
  }

  try {
    switch (event.type) {
      case "checkout.session.completed": {
        const cs = event.data.object as Stripe.Checkout.Session;
        const userId = cs.metadata?.userId;
        const plan = cs.metadata?.plan === "PRO" ? "PRO" : "PREMIUM";
        if (userId) {
          // mode subscription: l'estat real ve de la subscription de Stripe
          let subId = typeof cs.subscription === "string" ? cs.subscription : cs.subscription?.id || null;
          // v22: sincronitza TOTS els camps del encàrrec (període, interval,
          // cancelAtPeriodEnd, price) directament des de la font de veritat
          let stripeData: ReturnType<typeof subscriptionLocalData> | null = null;
          if (subId) {
            try {
              const sub = await stripe.subscriptions.retrieve(subId);
              stripeData = subscriptionLocalData(sub);
            } catch {
              // si no es pot llegir, el següent subscription.updated ho corregirà
            }
          }
          await db.user.update({ where: { id: userId }, data: { plan } });
          // P1.4: una ÚNICA subscripció activa per usuari — les anteriors
          // queden tancades (l'upsert de sota reactiva la de subId)
          await db.subscription.updateMany({
            where: { userId, status: "active" },
            data: { status: "canceled" },
          });
          if (subId) {
            await db.subscription.upsert({
              where: { stripeSubscriptionId: subId },
              create: {
                userId,
                stripeCustomerId: String(cs.customer || ""),
                stripeSubscriptionId: subId,
                plan,
                ...(stripeData ?? {
                  status: "active",
                  currentPeriodStart: null,
                  currentPeriodEnd: null,
                  cancelAtPeriodEnd: false,
                  billingInterval: null,
                  stripePriceId: null,
                }),
              },
              update: { plan, ...(stripeData ?? {}) },
            });
          } else {
            await db.subscription.create({
              data: {
                userId,
                stripeCustomerId: String(cs.customer || ""),
                stripeSubscriptionId: `session_${cs.id}`,
                plan,
                status: "active",
                currentPeriodEnd: null,
              },
            });
          }
          await db.event
            .create({ data: { type: "SUBSCRIBE_SUCCESS", userId, meta: JSON.stringify({ plan, mode: "stripe" }) } })
            .catch(() => {});
          await audit(userId, "STRIPE_SUBSCRIBE", subId || cs.id, { plan, status: stripeData?.status ?? "active" });
          const u = await db.user.findUnique({ where: { id: userId } });
          if (u) {
            const locale = u.locale === "es" ? "es" : "ca";
            const amount = plan === "PRO" ? "19,99 €" : "7,99 €";
            const mail = receiptMail(plan === "PRO" ? "Pro" : "Premium", amount, locale);
            after(async () => {
              await sendMail({
                to: u.email,
                subject: mail.subject,
                html: mail.html,
                type: "RECEIPT",
                userId: u.id,
                // P1 v19 — idempotència: re-entregues del webhook o reprocessos
                // mai generen un segon rebut per la mateixa checkout session
                idempotencyKey: `stripe-receipt:${subId || cs.id}`,
              });
            });
          }
        }
        break;
      }
      // v22: `created` també es gestiona — una subscripció creada fora del
      // checkout (p. ex. des del Portal de Stripe) queda sincronitzada; l'usuari
      // es dedueix del customer a partir de subscripcions anteriors.
      case "customer.subscription.created":
      case "customer.subscription.updated": {
        const s = event.data.object as Stripe.Subscription;
        const data = subscriptionLocalData(s);
        let row = await db.subscription.findUnique({ where: { stripeSubscriptionId: String(s.id) } });
        if (!row && event.type === "customer.subscription.created") {
          const prior = await db.subscription.findFirst({
            where: { stripeCustomerId: String(s.customer || "") },
            orderBy: { createdAt: "desc" },
          });
          if (prior) {
            row = await db.subscription.create({
              data: {
                userId: prior.userId,
                stripeCustomerId: String(s.customer || ""),
                stripeSubscriptionId: String(s.id),
                plan: prior.plan === "PRO" ? "PRO" : "PREMIUM",
                ...data,
              },
            });
          }
        }
        if (row) {
          await db.subscription.update({ where: { id: row.id }, data });
          // P1.4: el pla de l'usuari es recalcula SEMPRE a partir de les
          // subscripcions vigents (mai es confia en un estat intermedi)
          await syncUserPlan(row.userId, { force: true });
          const stillPaid = data.status === "active" || data.status === "trialing";
          if (!stillPaid) {
            await audit(null, "STRIPE_SUB_DOWNGRADED", String(s.id), { status: data.status });
          }
        }
        break;
      }
      case "customer.subscription.deleted": {
        const s = event.data.object as Stripe.Subscription;
        const sub = await db.subscription.findUnique({ where: { stripeSubscriptionId: String(s.id) } });
        if (sub) {
          await db.subscription.update({
            where: { id: sub.id },
            data: { status: "canceled", cancelAtPeriodEnd: false },
          });
          // P1.4: el pla de l'usuari es recalcula SEMPRE a partir de les
          // subscripcions vigents (mai es confia en un estat intermedi)
          await syncUserPlan(sub.userId, { force: true });
          await audit(null, "STRIPE_SUB_DOWNGRADED", String(s.id), { status: "canceled" });
        }
        break;
      }
      case "invoice.paid": {
        const inv = event.data.object as Stripe.Invoice;
        const customerId = String(inv.customer || "");
        const sub = await db.subscription.findFirst({ where: { stripeCustomerId: customerId } });
        if (sub) {
          await db.subscription.update({
            where: { id: sub.id },
            data: {
              status: "active",
              currentPeriodEnd: inv.period_end ? new Date(inv.period_end * 1000) : sub.currentPeriodEnd,
            },
          });
          await syncUserPlan(sub.userId, { force: true }); // P1.4: reactivació coherent
        }
        break;
      }
      case "invoice.payment_failed": {
        const inv = event.data.object as Stripe.Invoice;
        const customerId = String(inv.customer || "");
        const sub = await db.subscription.findFirst({ where: { stripeCustomerId: customerId } });
        if (sub) {
          await db.subscription.update({ where: { id: sub.id }, data: { status: "past_due" } });
          await audit(null, "STRIPE_PAYMENT_FAILED", String(inv.id), { customerId });
        }
        break;
      }
    }
  } catch (e) {
    // P1.1: alliberem el lock → Stripe reentregarà i l'event es reprocessarà
    await db.stripeEvent.delete({ where: { eventId: event.id } }).catch(() => {});
    console.error("stripe webhook", e);
    return NextResponse.json({ error: "PROCESSING_FAILED" }, { status: 500 });
  }
  return NextResponse.json({ received: true });
}

"use client";

import { useCallback, useEffect, useState } from "react";

/**
 * v22 — useUserPlan() (encàrrec Fase 1, punt 6).
 *
 * Obté l'estat de la suscripció DES DEL SERVIDOR (GET /api/subscription) i mai
 * des del client: el pla efectiu el decideix el backend a partir de les
 * subscripcions vigents (equivalent a supabase.auth.getUser() + RLS del
 * encàrrec). El frontend es refresca amb `refresh()` després de qualsevol
 * acció que canvii la suscripció (checkout, cancelar, reactivar...).
 *
 * Camps retornats (exactament els del encàrrec): planType, subscriptionStatus,
 * isFree, isPremium, isProAgents, hasPremiumAccess, hasProAgentsAccess,
 * currentPeriodEnd, cancelAtPeriodEnd, subscriptionCount,
 * monthlyDetailedQueriesUsed, loading.
 */

export interface UserPlanState {
  planType: "free" | "premium" | "pro_agents";
  subscriptionStatus: string;
  isFree: boolean;
  isPremium: boolean;
  isProAgents: boolean;
  hasPremiumAccess: boolean;
  hasProAgentsAccess: boolean;
  currentPeriodStart: string | null;
  currentPeriodEnd: string | null;
  cancelAtPeriodEnd: boolean;
  subscriptionCount: number;
  monthlyDetailedQueriesUsed: number;
  stripeConfigured: boolean;
  demoCheckout: boolean;
}

const DEFAULT_STATE: UserPlanState = {
  planType: "free",
  subscriptionStatus: "free",
  isFree: true,
  isPremium: false,
  isProAgents: false,
  hasPremiumAccess: false,
  hasProAgentsAccess: false,
  currentPeriodStart: null,
  currentPeriodEnd: null,
  cancelAtPeriodEnd: false,
  subscriptionCount: 0,
  monthlyDetailedQueriesUsed: 0,
  stripeConfigured: false,
  demoCheckout: false,
};

export function useUserPlan() {
  const [state, setState] = useState<UserPlanState>(DEFAULT_STATE);
  const [loading, setLoading] = useState(true);

  const refresh = useCallback(async () => {
    try {
      const r = await fetch("/api/subscription", { cache: "no-store" });
      if (r.ok) {
        const d = await r.json();
        setState({ ...DEFAULT_STATE, ...d });
      } else if (r.status === 401) {
        // sense sessió: estat per defecte (pla gratuït)
        setState(DEFAULT_STATE);
      }
      // altres errors: es manté l'últim estat conegut
    } catch {
      // xarxa caiguda: es manté l'últim estat conegut
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    refresh();
  }, [refresh]);

  return { ...state, loading, refresh };
}
