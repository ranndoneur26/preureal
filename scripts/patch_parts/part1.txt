export type PlanType = "free" | "premium" | "pro_agents";

const PLAN_RANK: Record<PlanId, number> = { FREE: 0, PREMIUM: 1, PRO: 2 };

export const USAGE_TYPES = {
  DETAIL_QUERY: "DETAIL_QUERY",
  EXPORT_CSV: "EXPORT_CSV",
  PDF_REPORT: "PDF_REPORT",
} as const;
export type UsageType = (typeof USAGE_TYPES)[keyof typeof USAGE_TYPES];

/** Clau de mes UTC: "2026-11". */
export function monthKey(d: Date = new Date()): string {
  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`;
}

export function planIdToType(plan: PlanId): PlanType {
  return plan === "PRO" ? "pro_agents" : plan === "PREMIUM" ? "premium" : "free";
}

/** Tradueix el planType del encàrrec (free|premium|pro_agents) al PlanId intern. */
export function typeToPlanId(t: string): "PREMIUM" | "PRO" | null {
  const v = String(t || "").toLowerCase();
  if (v === "premium") return "PREMIUM";
  if (v === "pro_agents" || v === "pro") return "PRO";
  return null;
}

/**
 * Comptador d'ús mensual (upsert atòmic per la clau única userId+type+month).
 * MAI llança: un fallo de telemetria no pot trencar mai l'acció de l'usuari.
 */
export async function recordUsage(userId: string, type: UsageType, n: number = 1): Promise<number | null> {
  try {
    const row = await db.subscriptionUsage.upsert({
      where: {
        userId_usageType_usageMonth: { userId, usageType: type, usageMonth: monthKey() },
      },
      create: { userId, usageType: type, usageMonth: monthKey(), usageCount: n },
      update: { usageCount: { increment: n } },
    });
    return row.usageCount;
  } catch (e) {
    console.error("recordUsage", type, e);
    return null;
  }
}

/** Dades locals que es deriven d'una subscripció de Stripe (font de veritat). */
export function subscriptionLocalData(s: Stripe.Subscription) {
  const item = s.items.data[0];
  const interval = item?.price?.recurring?.interval;
  return {
    status: s.status,
    currentPeriodStart: item?.current_period_start ? new Date(item.current_period_start * 1000) : null,
    currentPeriodEnd: item?.current_period_end ? new Date(item.current_period_end * 1000) : null,
    cancelAtPeriodEnd: Boolean(s.cancel_at_period_end),
    billingInterval: interval === "month" || interval === "year" ? interval : null,
    stripePriceId: item?.price?.id ?? null,
  };
}

/** El pla més alt de les subscripcions vigents (defensiu; requireUser ja sincronitza). */
export function highestVigentPlan(
  subs: Array<{ status: string; plan: string }>
): PlanId {
  let plan: PlanId = "FREE";
  for (const s of subs) {
    if (s.status !== "active" && s.status !== "trialing") continue;
    const p = (s.plan === "PREMIUM" || s.plan === "PRO" ? s.plan : "FREE") as PlanId;
    if (PLAN_RANK[p] > PLAN_RANK[plan]) plan = p;
  }
  return plan;
}

export interface SubscriptionState {

  subscriptionStatus: string;
  isFree: boolean;
  isPremium: boolean;
  isProAgents: boolean;
  hasPremiumAccess: boolean;
  hasProAgentsAccess: boolean;
  currentPeriodStart: string | null;
  currentPeriodEnd: string | null;
  cancelAtPeriodEnd: boolean;
  subscriptionCount: number;
  monthlyDetailedQueriesUsed: number;
  stripeConfigured: boolean;
  demoCheckout: boolean;
}

/**
 * Estat complet per al hook useUserPlan(). `user` ve de requireUser() (pla
 * fresc, ja sincronitzat amb les subscripcions vigents i períodes vençuts).
 */
export async function getSubscriptionState(user: {
  id: string;
  plan: string;
}): Promise<SubscriptionState> {
  const [subs, usage] = await Promise.all([
    db.subscription.findMany({ where: { userId: user.id }, orderBy: { createdAt: "desc" } }),
    db.subscriptionUsage.findUnique({
      where: {
        userId_usageType_usageMonth: {
          userId: user.id,
          usageType: USAGE_TYPES.DETAIL_QUERY,
          usageMonth: monthKey(),
        },
      },
    }),
  ]);

  const plan = (user.plan === "PREMIUM" || user.plan === "PRO" ? user.plan : "FREE") as PlanId;
  const vigent =
    subs.find((s) => s.status === "active" || s.status === "trialing") ?? null;

  return {
    planType: planIdToType(plan),
    subscriptionStatus: vigent?.status ?? "free",
    isFree: plan === "FREE",
    isPremium: plan === "PREMIUM",
    isProAgents: plan === "PRO",
    hasPremiumAccess: PLAN_RANK[plan] >= PLAN_RANK.PREMIUM,
    hasProAgentsAccess: PLAN_RANK[plan] >= PLAN_RANK.PRO,
    currentPeriodStart: vigent?.currentPeriodStart?.toISOString() ?? null,
    currentPeriodEnd: vigent?.currentPeriodEnd?.toISOString() ?? null,
    cancelAtPeriodEnd: vigent?.cancelAtPeriodEnd ?? false,
    subscriptionCount: subs.length,
    monthlyDetailedQueriesUsed: usage?.usageCount ?? 0,
    stripeConfigured: Boolean(process.env.STRIPE_SECRET_KEY),
    demoCheckout: process.env.DEMO_CHECKOUT === "true",
  };
}

/** Configuració efectiva dels plans: BD (PlanSetting) amb fallback als defaults. */
export async function getPlanSettings(): Promise<Record<PlanId, EffectivePlan>> {
  if (settingsCache && Date.now() - settingsCache.at < CACHE_MS) return settingsCache.data;
  const out: Record<PlanId, EffectivePlan> = {
    FREE: { ...PLANS.FREE, features: { ...PLANS.FREE.features } },
    PREMIUM: { ...PLANS.PREMIUM, features: { ...PLANS.PREMIUM.features } },
    PRO: { ...PLANS.PRO, features: { ...PLANS.PRO.features } },
  };
  try {
    const rows = await db.planSetting.findMany();
    for (const row of rows) {
      if (row.plan in out) {
        const id = row.plan as PlanId;
        out[id].quotaMonth = row.quotaMonth;
        out[id].priceEur = row.priceEur;
        if (row.features) {
          try {
            out[id].features = { ...out[id].features, ...JSON.parse(row.features) };
          } catch {}
        }
      }
    }
    settingsCache = { at: Date.now(), data: out };
  } catch (e) {
    console.error("getPlanSettings: BD no disponible, s'usen els defaults", e);
  }
  return out;
}

export function invalidatePlanCache() {
  settingsCache = null;
}

export function planName(plan: PlanId, locale: "ca" | "es"): string {
  const names: Record<PlanId, [string, string]> = {
    FREE: ["Gratuït", "Gratuito"],
    PREMIUM: ["Premium", "Premium"],
    PRO: ["Pro (agents)", "Pro (agentes)"],
  };
  return names[plan][locale === "es" ? 1 : 0];
}

/** Quota del pla segons la configuració efectiva. */
export async function quotaForPlan(plan: PlanId): Promise<number> {
  return (await getPlanSettings())[plan].quotaMonth;

/** true si el usuario tiene la consulta gratuita disponible este mes */
export async function canConsult(
  userId: string,
  plan: PlanId
): Promise<{ ok: boolean; used: number; limit: number }> {
  const limit = (await getPlanSettings())[plan].quotaMonth;
  if (limit < 0) return { ok: true, used: 0, limit: -1 };
  const start = new Date();
  start.setUTCDate(1);
  start.setUTCHours(0, 0, 0, 0);
  const used = await db.searchLog.count({
    where: { userId, kind: "DETAIL", createdAt: { gte: start } },
  });
  return { ok: used < limit, used, limit };
}

/** Comprova una feature del pla contra la configuració efectiva. */
export async function planAllows(plan: PlanId, feature: keyof EffectivePlan["features"]): Promise<boolean> {
  const s = await getPlanSettings();
  return Boolean(s[plan]?.features[feature]);
}

// Cau de sincronització per usuari (P1.4): evita martellejar la BD a cada petició
const planSyncAt = new Map<string, number>();
const PLAN_SYNC_MS = 60_000;

/**
 * P1.4 — Sincronitza el pla de l'usuari amb les seves subscripcions VIGENTS:
 *  1) expira les subscripcions "active" el període de les quals ja ha acabat
 *     (caducitat automàtica dels 30 dies DEMO o del període Stripe, sense
 *     dependre del botó de cancel·lar ni de cap cron);
 *  2) el pla efectiu és el MÉS ALT de les subscripcions encara actives
 *     (mai queden subscripcions antigues "actives" decidint el pla).
 *
 * Auto-curatiu: es crida des de requireUser() dels usuaris de pagament, des
 * dels webhooks de Stripe i des del flux DEMO. Mai llança: si la BD falla
 * retorna null i el caller manté el pla actual.
 */
export async function syncUserPlan(
  userId: string,
  opts?: { force?: boolean }
): Promise<PlanId | null> {
  try {
    if (!opts?.force) {
      const last = planSyncAt.get(userId);
      if (last && Date.now() - last < PLAN_SYNC_MS) return null; // sincronitzat fa poc
    }
    planSyncAt.set(userId, Date.now());

    const now = new Date();
    await db.subscription.updateMany({
      where: { userId, status: "active", currentPeriodEnd: { lt: now } },
      data: { status: "expired" },
    });
    const vigents = await db.subscription.findMany({
      where: { userId, status: { in: ["active", "trialing"] } },
      select: { plan: true },
    });
    let plan: PlanId = "FREE";
    for (const s of vigents) {
      const p = (s.plan === "PREMIUM" || s.plan === "PRO" ? s.plan : "FREE") as PlanId;
      if (PLAN_RANK[p] > PLAN_RANK[plan]) plan = p;
    }
    const user = await db.user.findUnique({ where: { id: userId }, select: { plan: true } });
    if (user && user.plan !== plan) {
      await db.user.update({ where: { id: userId }, data: { plan } });
    }
    return plan;
  } catch (e) {
    console.error("syncUserPlan", e);
    return null;
  }
}

function safeDecode(s: string): string {

async function dispatch(req: NextRequest, method: Method, parts: string[]): Promise<Response> {
  for (const r of TABLE) {
    if (r.seg.length !== parts.length) continue;
    const params: Record<string, string> = {};
    let ok = true;
    for (let i = 0; i < r.seg.length; i++) {
      const s = r.seg[i];
      const got = safeDecode(parts[i]);
      if (s.startsWith(":")) {
        params[s.slice(1)] = got;
      } else if (s !== got) {
        ok = false;
        break;
      }
    }
    if (!ok) continue;
    const fn = r.h[method] as Handler | undefined;
    if (!fn) {
      return NextResponse.json({ error: "method_not_allowed" }, { status: 405 });
    }
    return fn(req, { params: Promise.resolve(params) });
  }
  return NextResponse.json({ error: "not_found" }, { status: 404 });
}
