     1→"use client";
     2→
     3→import { useCallback, useEffect, useState } from "react";
     4→import { useApp } from "@/components/app-context";
     5→import { User, CreditCard, History, Star, GitCompare, FileText, Shield, Trash2, Settings, BadgeCheck, MailWarning, Download, ExternalLink, Search } from "lucide-react";
     6→
     7→interface BillingData {
     8→  subscriptions: Array<{ id: string; plan: string; status: string; demo: boolean; currentPeriodEnd: string | null; createdAt: string }>;
     9→  invoices: Array<{ id: string; date: string | null; amount: number | null; currency: string | null; status: string | null; hostedUrl: string | null }>;
    10→  stripeConfigured: boolean;
    11→}
    12→
    13→export default function PanelView() {
    14→  const { t, auth, navigate, locale, setLocale } = useApp();
    15→  const [tab, setTab] = useState("resum");
    16→  const [busy, setBusy] = useState(false);
    17→  const [msg, setMsg] = useState<string | null>(null);
    18→  const [verificacio, setVerificacio] = useState<string | null>(null);
    19→
    20→  // dades
    21→  const [name, setName] = useState("");
    22→  const [history, setHistory] = useState<any[] | null>(null);
    23→  const [favorites, setFavorites] = useState<any[] | null>(null);
    24→  const [comparisons, setComparisons] = useState<any[] | null>(null);
    25→  const [billing, setBilling] = useState<BillingData | null>(null);
    26→  const [reports, setReports] = useState<any[] | null>(null);
    27→  const [myReports, setMyReports] = useState<any[] | null>(null);
    28→
    29→  // seguretat
    30→  const [currentPassword, setCurrentPassword] = useState("");
    31→  const [newPassword, setNewPassword] = useState("");
    32→  const [deleteConfirm, setDeleteConfirm] = useState("");
    33→  const [activity, setActivity] = useState<any[] | null>(null);
    34→  const [lastLoginAt, setLastLoginAt] = useState<string | null>(null);
    35→
    36→  useEffect(() => {
    37→    auth.refresh();
    38→    const hq = typeof window !== "undefined" ? window.location.hash.split("?")[1] || "" : "";
    39→    if (hq) {
    40→      const p = new URLSearchParams(hq);
    41→      if (p.get("verificacio")) setVerificacio(p.get("verificacio"));
    42→    }
    43→     
    44→  }, []);
    45→
    46→  useEffect(() => {
    47→    if (auth.user) setName(auth.user.name || "");
    48→  }, [auth.user]);
    49→
    50→  const loadTab = useCallback(async (tabId: string) => {
    51→    const j = (u: string) => fetch(u).then((r) => (r.ok ? r.json() : null)).catch(() => null);
    52→    if (tabId === "historial" && history === null) {
    53→      const d = await j("/api/history");
    54→      if (d) setHistory(d.history);
    55→      else setHistory([]);
    56→    }
    57→    if (tabId === "preferits" && favorites === null) {
    58→      const d = await j("/api/favorites");
    59→      if (d) setFavorites(d.favorites);
    60→      else setFavorites([]);
    61→    }
    62→    if (tabId === "comparacions" && comparisons === null) {
    63→      const d = await j("/api/comparisons");
    64→      if (d) setComparisons(d.comparisons);
    65→      else setComparisons([]);
    66→    }
    67→    if (tabId === "facturacio" && billing === null) {
    68→      const d = await j("/api/billing/history");
    69→      if (d) setBilling(d);
    70→    }
    71→    if (tabId === "informes" && myReports === null) {
    72→      const d = await j("/api/account/export");
    73→      if (d) setMyReports(d.informesPreu || []);
    74→    }
    75→    if (tabId === "seguretat" && activity === null) {
    76→      const d = await j("/api/account/activity");
    77→      if (d) {
    78→        setActivity(d.events || []);
    79→        setLastLoginAt(d.lastLoginAt || null);
    80→      } else setActivity([]);
    81→    }
    82→  }, [history, favorites, comparisons, billing, myReports, activity]);
    83→
    84→  useEffect(() => { loadTab(tab); }, [tab, loadTab]);
    85→
    86→  if (!auth.user) {
    87→    return (
    88→      <div className="flex-1 flex items-center justify-center px-4">
    89→        <div className="text-center">
    90→          <p className="text-[#6b6f68] mb-4">{locale === "ca" ? "Has d'iniciar la sessió" : "Debes iniciar sesión"}</p>
    91→          <button onClick={() => navigate("/entrada")} className="px-5 py-2.5 rounded-xl bg-[#e8673c] text-white font-bold">
    92→            {t.nav.entrada}
    93→          </button>
    94→        </div>
    95→      </div>
    96→    );
    97→  }
    98→
    99→  const u = auth.user;
   100→  const planName =
   101→    u.plan === "PRO" ? t.plans.pro.name : u.plan === "PREMIUM" ? t.plans.premium.name : t.plans.free.name;
   102→
   103→  const saveProfile = async () => {
   104→    setBusy(true);
   105→    setMsg(null);
   106→    try {
   107→      const res = await fetch("/api/account", {
   108→        method: "PATCH",
   109→        headers: { "Content-Type": "application/json" },
   110→        body: JSON.stringify({ name, locale }),
   111→      });
   112→      if (!res.ok) {
   113→        const d = await res.json().catch(() => ({}));
   114→        setMsg(d.error || "Error");
   115→        return;
   116→      }
   117→      await auth.refresh();
   118→      setMsg(locale === "ca" ? "Dades desades." : "Datos guardados.");
   119→    } finally {
   120→      setBusy(false);
   121→    }
   122→  };
   123→
   124→  const changePassword = async () => {
   125→    setBusy(true);
   126→    setMsg(null);
   127→    try {
   128→      const res = await fetch("/api/auth/change-password", {
   129→        method: "POST",
   130→        headers: { "Content-Type": "application/json" },
   131→        body: JSON.stringify({ currentPassword, newPassword }),
   132→      });
   133→      const d = await res.json().catch(() => ({}));
   134→      if (!res.ok) {
   135→        setMsg(d.error === "PASSWORD_SHORT"
   136→          ? (locale === "ca" ? "La contrasenya nova és massa curta (mínim 8)." : "La contraseña nueva es demasiado corta (mín. 8).")
   137→          : d.error === "BAD_CREDENTIALS"
   138→          ? (locale === "ca" ? "La contrasenya actual no és correcta." : "La contraseña actual no es correcta.")
   139→          : "Error");
   140→        return;
   141→      }
   142→      setCurrentPassword("");
   143→      setNewPassword("");
   144→      setMsg(locale === "ca" ? "Contrasenya canviada. Les altres sessions s'han tancat." : "Contraseña cambiada. Las otras sesiones se han cerrado.");
   145→    } finally {
   146→      setBusy(false);
   147→    }
   148→  };
   149→
   150→  const resendVerification = async () => {
   151→    setBusy(true);
   152→    try {
   153→      const res = await fetch("/api/auth/verify", { method: "POST" });
   154→      setMsg(res.ok
   155→        ? (locale === "ca" ? "Correu de verificació enviat (revisa també el sandbox del panell admin si els correus no estan actius)." : "Correo de verificación enviado.")
   156→        : "Error");
   157→    } finally {
   158→      setBusy(false);
   159→    }
   160→  };
   161→
   162→  const revokeOtherSessions = async () => {
   163→    setBusy(true);
   164→    setMsg(null);
   165→    try {
   166→      const res = await fetch("/api/auth/sessions/revoke-all", { method: "POST" });
   167→      if (!res.ok) {
   168→        setMsg(locale === "ca" ? "No s'han pogut tancar les sessions." : "No se han podido cerrar las sesiones.");
   169→        return;
   170→      }
   171→      await auth.refresh();
   172→      setActivity(null); // recarrega l'activitat la propera vegada
   173→      setMsg(locale === "ca"
   174→        ? "S'han tancat les sessions d'altres dispositius. Aquesta sessió continua activa."
   175→        : "Se han cerrado las sesiones de otros dispositivos. Esta sesión sigue activa.");
   176→    } finally {
   177→      setBusy(false);
   178→    }
   179→  };
   180→
   181→  const cancelDemo = async () => {
   182→    setBusy(true);
   183→    await fetch("/api/stripe/demo", { method: "POST" });
   184→    await auth.refresh();
   185→    setBusy(false);
   186→  };
   187→
   188→  const openPortal = async () => {
   189→    setBusy(true);
   190→    try {
   191→      const res = await fetch("/api/stripe/portal", { method: "POST" });
   192→      const d = await res.json();
   193→      if (d.url) window.location.href = d.url;
   194→      else setMsg(d.message || d.error || "Error");
   195→    } finally {
   196→      setBusy(false);
   197→    }
   198→  };
   199→
   200→  const deleteAccount = async () => {
   201→    if (deleteConfirm !== "ESBORRA") {
   202→      setMsg(locale === "ca" ? "Escriu ESBORRA al camp de confirmació." : "Escribe ESBORRA en el campo de confirmación.");
   203→      return;
   204→    }
   205→    setBusy(true);
   206→    try {
   207→      const res = await fetch("/api/account", {
   208→        method: "DELETE",
   209→        headers: { "Content-Type": "application/json" },
   210→        body: JSON.stringify({ confirm: deleteConfirm }),
   211→      });
   212→      if (!res.ok) {
   213→        const d = await res.json().catch(() => ({}));
   214→        setMsg(d.error || "Error");
   215→        return;
   216→      }
   217→      navigate("/");
   218→      window.location.reload();
   219→    } finally {
   220→      setBusy(false);
   221→    }
   222→  };
   223→
   224→  const deleteHistory = async (id?: string) => {
   225→    const qs = id ? `id=${id}` : "all=1";
   226→    await fetch(`/api/history?${qs}`, { method: "DELETE" });
   227→    setHistory(null);
   228→    loadTab("historial");
   229→  };
   230→
   231→  const tabs = [
   232→    ["resum", t.panel.title, User],
   233→    ["perfil", locale === "ca" ? "Dades personals" : "Datos personales", Settings],
   234→    ["facturacio", locale === "ca" ? "Facturació" : "Facturación", CreditCard],
   235→    ["historial", locale === "ca" ? "Historial" : "Historial", History],
   236→    ["preferits", locale === "ca" ? "Preferits" : "Favoritos", Star],
   237→    ["comparacions", locale === "ca" ? "Comparacions" : "Comparaciones", GitCompare],
   238→    ["informes", locale === "ca" ? "Informes meus" : "Mis informes", FileText],
   239→    ["seguretat", locale === "ca" ? "Seguretat" : "Seguridad", Shield],
   240→  ] as const;
   241→
   242→  return (
   243→    <div className="flex-1 max-w-3xl w-full mx-auto px-4 py-12">
   244→      <h1 className="text-3xl font-extrabold">{t.panel.title}</h1>
   245→
   246→      {verificacio && (
   247→        <div className={`mt-4 rounded-xl p-4 text-sm ${verificacio === "ok" ? "bg-[#e6f4ea] text-[#1e7a41]" : "bg-[#fdeae2] text-[#b0431c]"}`}>
   248→          {verificacio === "ok" &&
   249→            (locale === "ca"
   250→              ? "Correu verificat correctament. Ja pots fer consultes detallades!"
   251→              : "¡Correo verificado correctamente. Ya puedes hacer consultas detalladas!")}
   252→          {verificacio === "ja_usat" &&
   253→            (locale === "ca"
   254→              ? "Aquest enllaç de verificació ja s'havia utilitzat. Si el teu correu ja consta com a verificat, no cal fer res més."
   255→              : "Este enlace de verificación ya se había utilizado. Si tu correo ya consta como verificado, no hace falta nada más.")}
   256→          {verificacio === "caducat" &&
   257→            (locale === "ca"
   258→              ? "L'enllaç ha caducat (vàlid 24 hores). Reenvia'l i prova de nou."
   259→              : "El enlace ha caducado (válido 24 horas). Reenvíalo e inténtalo de nuevo.")}
   260→          {verificacio === "invalid" &&
   261→            (locale === "ca"
   262→              ? "L'enllaç de verificació no és vàlid."
   263→              : "El enlace de verificación no es válido.")}
   264→          {verificacio !== "ok" && (
   265→            <button onClick={resendVerification} disabled={busy} className="ml-1 underline font-semibold disabled:opacity-50">
   266→              {locale === "ca" ? "Reenviar correu de verificació" : "Reenviar correo de verificación"}
   267→            </button>
   268→          )}
   269→        </div>
   270→      )}
   271→      {msg && (
   272→        <div className="mt-4 rounded-xl bg-[#f0ede8] text-sm p-3">{msg}</div>
   273→      )}
   274→
   275→      {/* pestanyes */}
   276→      <div className="mt-6 flex gap-1 flex-wrap">
   277→        {tabs.map(([id, label, Icon]) => (
   278→          <button
   279→            key={id}
   280→            onClick={() => setTab(id)}
   281→            className={`px-3.5 py-2 rounded-lg text-sm font-semibold flex items-center gap-1.5 ${
   282→              tab === id ? "bg-[#0f2e21] text-white" : "bg-white border border-[#e5e1da] hover:bg-[#f0ede8]"
   283→            }`}
   284→          >
   285→            <Icon className="w-4 h-4" />
   286→            {label}
   287→          </button>
   288→        ))}
   289→      </div>
   290→
   291→      {/* RESUM */}
   292→      {tab === "resum" && (
   293→        <div className="mt-6 bg-white rounded-2xl border border-[#e5e1da] p-6 space-y-4">
   294→          <Row label={t.auth.email} value={u.email} />
   295→          <Row label={t.panel.plan} value={planName} strong />
   296→          {u.plan === "FREE" && auth.quota && (
   297→            <Row label={t.panel.quota} value={`${auth.quota.used} / ${auth.quota.limit}`} />
   298→          )}
   299→          {(u.plan === "PREMIUM" || u.plan === "PRO") && (
   300→            <>
   301→              <Row label={t.panel.quota} value={t.panel.unlimited} />
   302→              <Row
   303→                label={t.panel.subscription}
   304→                value={billing?.subscriptions.find((s) => s.demo) ? (locale === "ca" ? "DEMO (prova, sense cobrament)" : "DEMO (prueba, sin cobro)") : t.panel.active}
   305→              />
   306→            </>
   307→          )}
   308→          <div className="flex items-center justify-between gap-2">
   309→            <span className="text-sm text-[#6b6f68]">{locale === "ca" ? "Correu verificat" : "Correo verificado"}</span>
   310→            {u.emailVerified ? (
   311→              <span className="text-xs font-bold text-[#1e7a41] flex items-center gap-1"><BadgeCheck className="w-4 h-4" /> OK</span>
   312→            ) : (
   313→              <button onClick={resendVerification} disabled={busy} className="text-xs text-[#b0431c] flex items-center gap-1 underline disabled:opacity-50">
   314→                <MailWarning className="w-4 h-4" /> {locale === "ca" ? "No verificat — reenviar" : "No verificado — reenviar"}
   315→              </button>
   316→            )}
   317→          </div>
   318→          {u.plan === "FREE" && (
   319→            <div className="pt-2">
   320→              <button onClick={() => navigate("/preus")} className="px-5 py-2.5 rounded-xl bg-[#e8673c] text-white text-sm font-bold hover:bg-[#d95a31]">
   321→                {t.map.paywall.cta}
   322→              </button>
   323→            </div>
   324→          )}
   325→          {u.role === "ADMIN" && (
   326→            <div className="pt-2">
   327→              <button onClick={() => navigate("/admin")} className="px-5 py-2.5 rounded-xl bg-[#0f2e21] text-white text-sm font-bold hover:bg-[#16402e]">
   328→                {t.panel.adminLink}
   329→              </button>
   330→            </div>
   331→          )}
   332→          <div className="pt-2">
   333→            <button onClick={async () => { await fetch("/api/auth/logout", { method: "POST" }); await auth.refresh(); navigate("/"); }} className="text-sm text-[#6b6f68] hover:underline">
   334→              {t.panel.logout}
   335→            </button>
   336→          </div>
   337→        </div>
   338→      )}
   339→
   340→      {/* PERFIL */}
   341→      {tab === "perfil" && (
   342→        <div className="mt-6 bg-white rounded-2xl border border-[#e5e1da] p-6 space-y-4">
   343→          <div>
   344→            <label className="text-xs text-[#6b6f68]">{locale === "ca" ? "Nom (opcional)" : "Nombre (opcional)"}</label>
   345→            <input value={name} onChange={(e) => setName(e.target.value.slice(0, 80))} className="w-full mt-1 px-3 py-2 rounded-lg border border-[#e5e1da] text-sm" />
   346→          </div>
   347→          <div>
   348→            <label className="text-xs text-[#6b6f68]">{locale === "ca" ? "Idioma" : "Idioma"}</label>
   349→            <select value={locale} onChange={(e) => setLocale(e.target.value as "ca" | "es")} className="w-full mt-1 px-3 py-2 rounded-lg border border-[#e5e1da] text-sm">
   350→              <option value="ca">Català</option>
   351→              <option value="es">Castellano</option>
   352→            </select>
   353→          </div>
   354→          <button onClick={saveProfile} disabled={busy} className="px-5 py-2.5 rounded-xl bg-[#0f2e21] text-white text-sm font-bold disabled:opacity-50">
   355→            {locale === "ca" ? "Desar canvis" : "Guardar cambios"}
   356→          </button>
   357→          <div className="pt-2 border-t border-[#f0ede8]">
   358→            <a href="/api/account/export" className="text-sm text-[#0f2e21] underline flex items-center gap-1.5">
   359→              <Download className="w-4 h-4" /> {locale === "ca" ? "Descarrega les meves dades (RGPD)" : "Descarga mis datos (RGPD)"}
   360→            </a>
   361→          </div>
   362→        </div>
   363→      )}
   364→
   365→      {/* FACTURACIÓ */}
   366→      {tab === "facturacio" && (
   367→        <div className="mt-6 space-y-4">
   368→          <div className="bg-white rounded-2xl border border-[#e5e1da] p-6">
   369→            <h3 className="font-bold mb-3">{locale === "ca" ? "Gestió de subscripció" : "Gestión de suscripción"}</h3>
   370→            {billing === null ? (
   371→              <p className="text-sm text-[#6b6f68]">…</p>
   372→            ) : billing.subscriptions.length === 0 ? (
   373→              <p className="text-sm text-[#6b6f68]">{locale === "ca" ? "Encara no tens cap subscripció." : "Aún no tienes ninguna suscripción."}</p>
   374→            ) : (
   375→              <div className="space-y-2 text-sm">
   376→                {billing.subscriptions.map((s) => (
   377→                  <Row key={s.id} label={`${s.plan}${s.demo ? " (DEMO)" : ""}`} value={s.status} />
   378→                ))}
   379→              </div>
   380→            )}
   381→            <div className="mt-4 flex gap-2 flex-wrap">
   382→              {billing?.stripeConfigured && (
   383→                <button onClick={openPortal} disabled={busy} className="px-4 py-2 rounded-lg bg-[#0f2e21] text-white text-sm font-semibold flex items-center gap-1.5 disabled:opacity-50">
   384→                  <ExternalLink className="w-4 h-4" /> {locale === "ca" ? "Portal de client Stripe" : "Portal de cliente Stripe"}
   385→                </button>
   386→              )}
   387→              {(u.plan === "PREMIUM" || u.plan === "PRO") && billing?.subscriptions.some((s) => s.demo) && (
   388→                <button onClick={cancelDemo} disabled={busy} className="text-sm text-[#b0431c] underline disabled:opacity-50">
   389→                  {t.panel.cancelDemo}
   390→                </button>
   391→              )}
   392→            </div>
   393→          </div>
   394→          <div className="bg-white rounded-2xl border border-[#e5e1da] p-6">
   395→            <h3 className="font-bold mb-3">{locale === "ca" ? "Historial de pagaments i factures" : "Historial de pagos y facturas"}</h3>
   396→            {billing?.invoices.length ? (
   397→              <div className="space-y-2">
   398→                {billing.invoices.map((inv) => (
   399→                  <div key={inv.id} className="flex items-center justify-between gap-2 text-sm border border-[#f2efe9] rounded-lg px-3 py-2">
   400→                    <span>{inv.date ? new Date(inv.date).toLocaleDateString() : "—"} · {inv.amount?.toFixed(2)} {inv.currency?.toUpperCase()}</span>
   401→                    <span className="flex items-center gap-2">
   402→                      <span className="text-xs text-[#6b6f68]">{inv.status}</span>
   403→                      {inv.hostedUrl && <a href={inv.hostedUrl} target="_blank" rel="noreferrer" className="text-xs underline text-[#0f2e21]">{locale === "ca" ? "Factura" : "Factura"}</a>}
   404→                    </span>
   405→                  </div>
   406→                ))}
   407→              </div>
   408→            ) : (
   409→              <p className="text-sm text-[#6b6f68]">
   410→                {billing?.stripeConfigured
   411→                  ? (locale === "ca" ? "Cap factura encara." : "Ninguna factura todavía.")
   412→                  : (locale === "ca" ? "Les factures de Stripe apareixeran aquí quan el propietari connecti el compte de Stripe." : "Las facturas de Stripe aparecerán aquí cuando el propietario conecte la cuenta de Stripe.")}
   413→              </p>
   414→            )}
   415→          </div>
   416→        </div>
   417→      )}
   418→
   419→      {/* HISTORIAL */}
   420→      {tab === "historial" && (
   421→        <div className="mt-6 bg-white rounded-2xl border border-[#e5e1da] p-6">
   422→          <div className="flex items-center justify-between mb-3">
   423→            <h3 className="font-bold">{locale === "ca" ? "Historial de consultes (Premium)" : "Historial de consultas (Premium)"}</h3>
   424→            {history && history.length > 0 && (
   425→              <button onClick={() => deleteHistory()} className="text-xs text-[#b0431c] underline">{locale === "ca" ? "Buida-ho tot" : "Vaciar todo"}</button>
   426→            )}
   427→          </div>
   428→          {history === null ? (
   429→            <p className="text-sm text-[#6b6f68]">…</p>
   430→          ) : history.length === 0 ? (
   431→            <p className="text-sm text-[#6b6f68]">{locale === "ca" ? "Cap consulta registrada (el pla gratuït no desa historial)." : "Ninguna consulta registrada (el plan gratuito no guarda historial)."}</p>
   432→          ) : (
   433→            <div className="space-y-2">
   434→              {history.map((h) => (
   435→                <div key={h.id} className="flex items-center justify-between gap-2 text-sm border border-[#f2efe9] rounded-lg px-3 py-2">
   436→                  <span>
   437→                    <b>{h.municipalityName || h.municipalityCode}</b>

   526→          )}
   527→        </div>

   530→      {/* SEGURETAT */}
   531→      {tab === "seguretat" && (

   573→            <button onClick={changePassword} disabled={busy || !currentPassword || !newPassword} className="px-5 py-2.5 rounded-xl bg-[#0f2e21] text-white text-sm font-bold disabled:opacity-50">

   577→          </div>

   581→            <input value={deleteConfirm} onChange={(e) => setDeleteConfirm(e.target.value.toUpperCase())} placeholder={locale === "ca" ? 'Escriu "ESBORRA" per confirmar' : 'Escribe "ESBORRA" para confirmar'} className="w-full px-3 py-2 rounded-lg border border-[#f2c4ac] text-sm" />
   582→            <button onClick={deleteAccount} disabled={busy || deleteConfirm !== "ESBORRA"} className="px-5 py-2.5 rounded-xl bg-[#b0431c] text-white text-sm font-bold disabled:opacity-40">
   583→              {locale === "ca" ? "Elimina el meu compte definitivament" : "Elimina mi cuenta definitivamente"}
   584→            </button>
   585→          </div>
   586→        </div>
   587→      )}
   588→    </div>
   589→  );
   590→}
