    33→    33→  error?: string;

    35→    35→
    36→    36→function rentColor(rent: number | null | undefined): string {
    37→    37→  if (!rent) return "#9aa5a0";
    38→    38→  if (rent < 550) return "#63b57c";
    39→    39→  if (rent < 700) return "#93c47d";
    40→    40→  if (rent < 850) return "#e5c46a";
    41→    41→  if (rent < 1000) return "#e89a4b";
    42→    42→  return "#dd5a35";
    43→    43→}
    44→    44→
    45→    45→export default function MapView() {

    47→    47→  const mapDivRef = useRef<HTMLDivElement>(null);
    48→    48→  const mapRef = useRef<any>(null);
    49→    49→  const LRef = useRef<any>(null);
    50→    50→  const layerRef = useRef<any>(null);
    51→    51→  const boundaryRef = useRef<any>(null);
    52→    52→  const selectedRef = useRef<any>(null);
    53→    53→
    54→    54→  const [query, setQuery] = useState("");
    55→    55→  const [results, setResults] = useState<MunLite[]>([]);
    56→    56→  const [showResults, setShowResults] = useState(false);
    57→    57→  const [selectedMun, setSelectedMun] = useState<MunLite | null>(null);
    58→    58→  const [zoomHint, setZoomHint] = useState(false);
    59→    59→  const [loadingBuildings, setLoadingBuildings] = useState(false);
    60→    60→  const [buildingsError, setBuildingsError] = useState<string | null>(null);
    61→    61→  const [panelState, setPanelState] = useState<"closed" | "loading" | "teaser" | "full" | "paywall">("closed");
    62→    62→  const [lookup, setLookup] = useState<LookupResult | null>(null);
    63→    63→  const [quota, setQuota] = useState<{ used: number; limit: number } | null>(auth.quota);
    64→    64→  const [demoNotice, setDemoNotice] = useState<string | null>(null);
    65→    65→
    66→    66→  useEffect(() => { setQuota(auth.quota); }, [auth.quota]);
    67→    67→
    68→    68→  // ---- Inicialització del mapa ----
    69→    69→  useEffect(() => {
    70→    70→    let cancelled = false;
    71→    71→    (async () => {
    72→    72→      const L = await import("leaflet");
    73→    73→      if (cancelled) return;
    74→    74→      LRef.current = L;
    75→    75→      if (!mapDivRef.current || mapRef.current) return;
    76→    76→      const map = L.map(mapDivRef.current, {
    77→    77→        center: [41.3925, 2.1650],
    78→    78→        zoom: 16,
    79→    79→        zoomControl: true,
    80→    80→        attributionControl: true,
    81→    81→      });
    82→    82→      mapRef.current = map;
    83→    83→      L.tileLayer("https://geoserveis.icgc.cat/servei/catalunya/mapa-base/wmts/topografic-gris/MON3857NW/{z}/{x}/{y}.png", {
    84→    84→        attribution: 'Base: <a href="https://www.icgc.cat">ICGC</a> (CC-BY 4.0) · Edificis: ICGC RTT · Dades: Incasòl / habitatge.gencat.cat',
    85→    85→        maxZoom: 19,
    86→    86→      }).addTo(map);
    87→    87→      layerRef.current = L.layerGroup().addTo(map);
    88→    88→      boundaryRef.current = L.layerGroup().addTo(map);
    89→    89→
    90→    90→      const onMove = () => loadBuildingsInView();
    91→    91→      map.on("moveend", onMove);
    92→    92→      loadBuildingsInView();
    93→    93→    })();
    94→    94→    return () => {
    95→    95→      cancelled = true;
    96→    96→      if (mapRef.current) {
    97→    97→        mapRef.current.remove();
    98→    98→        mapRef.current = null;
    99→    99→      }
   100→   100→    };
   101→   101→     
   102→   102→  }, []);
   103→   103→
   104→   104→  const loadBuildingsInView = useCallback(async () => {
   105→   105→    const map = mapRef.current;
   106→   106→    if (!map || !layerRef.current) return;
   107→   107→    const z = map.getZoom();
   108→   108→    if (z < 16) {
   109→   109→      layerRef.current.clearLayers();
   110→   110→      setZoomHint(true);
   111→   111→      setLoadingBuildings(false);
   112→   112→      return;
   113→   113→    }
   114→   114→    setZoomHint(false);
   115→   115→    const b = map.getBounds();
   116→   116→    const qs = new URLSearchParams({
   117→   117→      west: String(b.getWest().toFixed(5)),
   118→   118→      south: String(b.getSouth().toFixed(5)),
   119→   119→      east: String(b.getEast().toFixed(5)),
   120→   120→      north: String(b.getNorth().toFixed(5)),
   121→   121→    });
   122→   122→    setLoadingBuildings(true);
   123→   123→    setBuildingsError(null);
   124→   124→    try {
   125→   125→      const res = await fetch(`/api/buildings?${qs}`);
   126→   126→      if (!res.ok) {
   127→   127→        const d = await res.json().catch(() => ({}));
   128→   128→        if (d.error === "BBOX_TOO_BIG") {
   129→   129→          layerRef.current.clearLayers();
   130→   130→          setBuildingsError("zoom");
   131→   131→        } else {
   132→   132→          setBuildingsError("service");
   133→   133→        }
   134→   134→        return;
   135→   135→      }
   136→   136→      const data = await res.json();
   137→   137→      drawBuildings(data.buildings || []);
   138→   138→    } catch {
   139→   139→      setBuildingsError("service");
   140→   140→    } finally {
   141→   141→      setLoadingBuildings(false);
   142→   142→    }
   143→   143→     
   144→   144→  }, []);
   145→   145→
   146→   146→  // recarrega edificis quan canvia l'idioma/sessió per refrescar colors i accés
   147→   147→  useEffect(() => {
   148→   148→    loadBuildingsInView();
   149→   149→     
   150→   150→  }, [locale]);
   151→   151→
   152→   152→  const drawBuildings = useCallback((buildings: any[]) => {
   153→   153→    const L = LRef.current;
   154→   154→    const layer = layerRef.current;
   155→   155→    if (!L || !layer) return;
   156→   156→    layer.clearLayers();
   157→   157→    for (const b of buildings) {
   158→   158→      let color = "#9aa5a0";
   159→   159→      if (b.munRent) color = rentColor(b.munRent);
   160→   160→      const poly = L.polygon(b.ring.map((p: number[]) => [p[1], p[0]]), {
   161→   161→        color: "#3d4a42",
   162→   162→        weight: 0.7,
   163→   163→        fillColor: color,
   164→   164→        fillOpacity: 0.55,
   165→   165→      });
   166→   166→      poly.on("click", (e: any) => {
   167→   167→        L.DomEvent.stopPropagation(e);
   168→   168→        onSelectBuilding(b, e.latlng.lat, e.latlng.lng, poly);
   169→   169→      });
   170→   170→      poly.addTo(layer);
   171→   171→    }
   172→   172→     
   173→   173→  }, []);
   174→   174→
   175→   175→  const lookupSeqRef = useRef(0);
   176→   176→
   177→   177→  const onSelectBuilding = useCallback(async (b: any, lat: number, lng: number, poly: any) => {
   178→   178→    const L = LRef.current;
   179→   179→    const map = mapRef.current;
   180→   180→    if (L && map) {
   181→   181→      if (selectedRef.current) {
   182→   182→        selectedRef.current.setStyle({ color: "#3d4a42", weight: 0.7, fillOpacity: 0.55 });
   183→   183→      }
   184→   184→      poly.setStyle({ color: "#e8673c", weight: 3, fillOpacity: 0.8 });
   185→   185→      selectedRef.current = poly;
   186→   186→    }
   187→   187→    // invalida respuestas de consultas anteriores (evita carreras fuera de orden)
   188→   188→    const seq = ++lookupSeqRef.current;
   189→   189→    setPanelState("loading");
   190→   190→    setLookup(null);
   191→   191→    setDemoNotice(null);
   192→   192→    try {
   193→   193→      const res = await fetch("/api/lookup", {
   194→   194→        method: "POST",
   195→   195→        headers: { "Content-Type": "application/json" },
   196→   196→        body: JSON.stringify({ lat, lng, buildingId: b.id }),
   197→   197→      });
   198→   198→      if (seq !== lookupSeqRef.current) return; // respuesta obsoleta
   199→   199→      if (res.status === 402) {
   200→   200→        const d = await res.json().catch(() => ({}));
   201→   201→        setLookup(d);
   202→   202→        setPanelState("paywall");
   203→   203→        return;
   204→   204→      }
   205→   205→      const d: LookupResult = await res.json();
   206→   206→      if (seq !== lookupSeqRef.current) return; // respuesta obsoleta
   207→   207→      setLookup(d);
   208→   208→      setPanelState(d.kind === "full" ? "full" : "teaser");
   209→   209→      if (d.kind === "full" && d.quota) setQuota(d.quota);
   210→   210→    } catch {
   211→   211→      if (seq === lookupSeqRef.current) setPanelState("closed");
   212→   212→    }
   213→   213→  }, []);
   214→   214→
   215→   215→  // ---- Cercador de municipis ----
   216→   216→  useEffect(() => {
   217→   217→    if (query.trim().length < 2) {
   218→   218→      setResults([]);
   219→   219→      return;
   220→   220→    }
   221→   221→    const id = setTimeout(async () => {
   222→   222→      try {
   223→   223→        const res = await fetch(`/api/municipalities?q=${encodeURIComponent(query.trim())}&limit=8`);
   224→   224→        const d = await res.json();
   225→   225→        setResults(d.municipalities || []);
   226→   226→        setShowResults(true);
   227→   227→      } catch {}
   228→   228→    }, 220);
   229→   229→    return () => clearTimeout(id);
   230→   230→  }, [query]);
   231→   231→
   232→   232→  const selectMunicipality = useCallback(async (mun: MunLite) => {
   233→   233→    setShowResults(false);
   234→   234→    setQuery(mun.name);
   235→   235→    setSelectedMun(mun);
   236→   236→    const map = mapRef.current;
   237→   237→    if (!map) return;
   238→   238→    // límit municipal real (ICGC amb cau) + fit
   239→   239→    try {
   240→   240→      const res = await fetch(`/api/municipalities/${mun.code}`);
   241→   241→      const d = await res.json();
   242→   242→      if (d.boundary) {
   243→   243→        const L = LRef.current;
   244→   244→        boundaryRef.current?.clearLayers();
   245→   245→        const layer = L.geoJSON(d.boundary, {
   246→   246→          style: { color: "#0f2e21", weight: 2, fill: false, dashArray: "4 3" },
   247→   247→        });
   248→   248→        layer.addTo(boundaryRef.current);
   249→   249→        map.fitBounds(layer.getBounds().pad(0.05));
   250→   250→        // si ja tenim prou zoom, carreguem edificis immediatament
   251→   251→        if (map.getZoom() >= 16) loadBuildingsInView();
   252→   252→      } else {
   253→   253→        map.setView([mun.lat, mun.lng], 16);
   254→   254→        setTimeout(() => loadBuildingsInView(), 400);
   255→   255→      }
   256→   256→    } catch {
   257→   257→      map.setView([mun.lat, mun.lng], 16);
   258→   258→    }
   259→   259→  }, [loadBuildingsInView]);
   260→   260→
   261→   261→  const doCheckout = async (plan: "PREMIUM" | "PRO") => {
   262→   262→    try {
   263→   263→      const res = await fetch("/api/stripe/checkout", {
   264→   264→        method: "POST",
   265→   265→        headers: { "Content-Type": "application/json" },
   266→   266→        body: JSON.stringify({ plan }),
   267→   267→      });
   268→   268→      const d = await res.json();
   269→   269→      if (d.ok && d.mode === "STRIPE" && d.url) {
   270→   270→        window.location.href = d.url;
   271→   271→      } else if (d.ok && d.mode === "DEMO") {
   272→   272→        setDemoNotice(d.message || "Demo");
   273→   273→        setPanelState("full");
   274→   274→        await auth.refresh();
   275→   275→      } else {
   276→   276→        setDemoNotice(d.message || d.error || "Error");
   277→   277→      }
   278→   278→    } catch {
   279→   279→      setDemoNotice("Error de xarxa");
   280→   280→    }
   281→   281→  };
   282→   282→
   283→   283→  return (
   284→   284→    <div className="flex-1 flex flex-col md:flex-row relative md:h-[calc(100vh-4rem)] md:overflow-hidden">
   285→   285→      {/* MAPA */}
   286→   286→      <div className="relative flex-1 min-h-[55vh] md:h-full md:min-h-[520px] md:flex-none md:basis-auto md:grow">
   287→   287→        <div ref={mapDivRef} className="absolute inset-0 z-0" />
   288→   288→
   289→   289→        {/* Cercador */}
   290→   290→        <div className="absolute top-3 left-3 right-3 z-[500] max-w-md">
   291→   291→          <div className="relative">
   292→   292→            <div className="bg-white rounded-xl shadow-lg border border-[#e5e1da] flex items-center px-3">
   293→   293→              <Search className="w-4 h-4 text-[#6b6f68] shrink-0" />
   294→   294→              <input
   295→   295→                value={query}
   296→   296→                onChange={(e) => setQuery(e.target.value)}
   297→   297→                onFocus={() => results.length && setShowResults(true)}
   298→   298→                onBlur={() => setTimeout(() => setShowResults(false), 180)}
   299→   299→                placeholder={t.map.searchPlaceholder}
   300→   300→                className="w-full px-2 py-3 text-sm outline-none bg-transparent"
   301→   301→                aria-label={t.map.searchPlaceholder}
   302→   302→              />
   303→   303→              {query && (
   304→   304→                <button onClick={() => { setQuery(""); setSelectedMun(null); boundaryRef.current?.clearLayers(); }}>
   305→   305→                  <X className="w-4 h-4 text-[#6b6f68]" />
   306→   306→                </button>
   307→   307→              )}
   308→   308→            </div>
   309→   309→            {showResults && results.length > 0 && (
   310→   310→              <div className="absolute mt-1 w-full bg-white rounded-xl shadow-xl border border-[#e5e1da] overflow-hidden max-h-80 overflow-y-auto pr-scroll">
   311→   311→                {results.map((m) => (
   312→   312→                  <button
   313→   313→                    key={m.code}
   314→   314→                    onMouseDown={(e) => e.preventDefault()}
   315→   315→                    onClick={() => selectMunicipality(m)}
   316→   316→                    className="w-full text-left px-4 py-2.5 hover:bg-[#f0ede8] flex items-center justify-between gap-2"
   317→   317→                  >
   318→   318→                    <span className="text-sm font-medium">{m.name}</span>
   319→   319→                    <span className="text-xs text-[#6b6f68] flex items-center gap-1.5">
   320→   320→                      {m.rentLatest ? (
   321→   321→                        <span className="font-bold text-[#0f2e21]">{Math.round(m.rentLatest)} €/mes</span>
   322→   322→                      ) : (
   323→   323→                        <span>—</span>
   324→   324→                      )}
   325→   325→                      {m.isTense && <Flame className="w-3.5 h-3.5 text-[#e8673c]" />}
   326→   326→                    </span>
   327→   327→                  </button>
   328→   328→                ))}
   329→   329→              </div>
   330→   330→            )}
   331→   331→          </div>
   332→   332→        </div>
   333→   333→
   334→   334→        {demoNotice && (
   335→   335→          <div className="absolute top-16 left-3 right-3 z-[500] max-w-md bg-[#0f2e21] text-white text-xs rounded-xl p-3 shadow-lg flex justify-between gap-2">
   336→   336→            <span>{demoNotice}</span>
   337→   337→            <button onClick={() => setDemoNotice(null)}><X className="w-3.5 h-3.5" /></button>
   338→   338→          </div>
   339→   339→        )}
   340→   340→
   341→   341→        {/* Estats */}
   342→   342→        <div className="absolute bottom-3 left-3 z-[500] flex flex-col gap-2 max-w-[85%]">

   344→   344→            <div className="bg-white/95 rounded-lg px-3 py-2 text-xs shadow border border-[#e5e1da] flex items-center gap-2">
   345→   345→              <Info className="w-3.5 h-3.5 text-[#e8673c]" /> {t.map.zoomHint}
   346→   346→            </div>
   347→   347→          )}
   348→   348→          {loadingBuildings && (
   349→   349→            <div className="bg-white/95 rounded-lg px-3 py-2 text-xs shadow border border-[#e5e1da] flex items-center gap-2">
   350→   350→              <span className="w-3 h-3 rounded-full border-2 border-[#e8673c] border-t-transparent animate-spin" />
   351→   351→              {t.map.loadingBuildings}
   352→   352→            </div>

   354→   354→          {buildingsError === "service" && (

   357→   357→                ? "El servei de cartografia no està disponible ara mateix. Torna-ho a provar."
   358→   358→                : "El servicio de cartografía no está disponible ahora mismo. Inténtalo de nuevo."}

   366→   366→          <div className="flex gap-1 items-center">
   367→   367→            {["#63b57c", "#93c47d", "#e5c46a", "#e89a4b", "#dd5a35"].map((c) => (

   370→   370→          </div>
   371→   371→          <div className="mt-1 flex items-center gap-1 text-[#6b6f68]">
