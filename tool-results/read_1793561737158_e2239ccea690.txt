     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import "leaflet/dist/leaflet.css";
     5→import { useApp } from "@/components/app-context";
     6→import { Flame, MapPin, Search, X, Building2, TrendingUp, TrendingDown, Minus, Info, ExternalLink } from "lucide-react";
     7→
     8→interface MunLite {
     9→  code: string;
    10→  name: string;
    11→  comarca: string | null;
    12→  lat: number;
    13→  lng: number;
    14→  rentLatest: number | null;
    15→  rentPeriod: string | null;
    16→  isTense: boolean;
    17→  tenseList?: string | null;
    18→}
    19→
    20→interface LookupResult {
    21→  kind: "full" | "teaser";
    22→  building?: { id?: string; refcat?: string | null; yearBuilt?: { year: number; source: string } | null; tipus?: string | null; altura?: number | null; nom?: string | null };
    23→  municipality?: {
    24→    code: string; name: string; comarca: string | null; province: string | null;
    25→    rentLatest: number | null; rentPeriod: string | null;
    26→    rentPrevYear: number | null; rentPrevPeriod: string | null;
    27→    rentHistory: Record<string, number> | null;
    28→    contractsLatest: number | null;
    29→    isTense: boolean; tenseList: string | null;
    30→  };
    31→  quota?: { used: number; limit: number };
    32→  teaser?: { hasRent: boolean; period: string | null; message: string };
    33→  error?: string;
    34→}
    35→
    36→function rentColor(rent: number | null | undefined): string {
    37→  if (!rent) return "#9aa5a0";
    38→  if (rent < 550) return "#63b57c";
    39→  if (rent < 700) return "#93c47d";
    40→  if (rent < 850) return "#e5c46a";
    41→  if (rent < 1000) return "#e89a4b";
    42→  return "#dd5a35";
    43→}
    44→
    45→export default function MapView() {
    46→  const { t, locale, auth, navigate } = useApp();
    47→  const mapDivRef = useRef<HTMLDivElement>(null);
    48→  const mapRef = useRef<any>(null);
    49→  const LRef = useRef<any>(null);
    50→  const layerRef = useRef<any>(null);
    51→  const boundaryRef = useRef<any>(null);
    52→  const selectedRef = useRef<any>(null);
    53→
    54→  const [query, setQuery] = useState("");
    55→  const [results, setResults] = useState<MunLite[]>([]);
    56→  const [showResults, setShowResults] = useState(false);
    57→  const [selectedMun, setSelectedMun] = useState<MunLite | null>(null);
    58→  const [zoomHint, setZoomHint] = useState(false);
    59→  const [loadingBuildings, setLoadingBuildings] = useState(false);
    60→  const [buildingsError, setBuildingsError] = useState<string | null>(null);
    61→  const [panelState, setPanelState] = useState<"closed" | "loading" | "teaser" | "full" | "paywall">("closed");
    62→  const [lookup, setLookup] = useState<LookupResult | null>(null);
    63→  const [quota, setQuota] = useState<{ used: number; limit: number } | null>(auth.quota);
    64→  const [demoNotice, setDemoNotice] = useState<string | null>(null);
    65→
    66→  useEffect(() => { setQuota(auth.quota); }, [auth.quota]);
    67→
    68→  // ---- Inicialització del mapa ----
    69→  useEffect(() => {
    70→    let cancelled = false;
    71→    (async () => {
    72→      const L = await import("leaflet");
    73→      if (cancelled) return;
    74→      LRef.current = L;
    75→      if (!mapDivRef.current || mapRef.current) return;
    76→      const map = L.map(mapDivRef.current, {
    77→        center: [41.3925, 2.1650],
    78→        zoom: 16,
    79→        zoomControl: true,
    80→        attributionControl: true,
    81→      });
    82→      mapRef.current = map;
    83→      L.tileLayer("https://geoserveis.icgc.cat/servei/catalunya/mapa-base/wmts/topografic-gris/MON3857NW/{z}/{x}/{y}.png", {
    84→        attribution: 'Base: <a href="https://www.icgc.cat">ICGC</a> (CC-BY 4.0) · Edificis: ICGC RTT · Dades: Incasòl / habitatge.gencat.cat',
    85→        maxZoom: 19,
    86→      }).addTo(map);
    87→      layerRef.current = L.layerGroup().addTo(map);
    88→      boundaryRef.current = L.layerGroup().addTo(map);
    89→
    90→      const onMove = () => loadBuildingsInView();
    91→      map.on("moveend", onMove);
    92→      loadBuildingsInView();
    93→    })();
    94→    return () => {
    95→      cancelled = true;
    96→      if (mapRef.current) {
    97→        mapRef.current.remove();
    98→        mapRef.current = null;
    99→      }
   100→    };
   101→     
   102→  }, []);
   103→
   104→  const loadBuildingsInView = useCallback(async () => {
   105→    const map = mapRef.current;
   106→    if (!map || !layerRef.current) return;
   107→    const z = map.getZoom();
   108→    if (z < 16) {
   109→      layerRef.current.clearLayers();
   110→      setZoomHint(true);
   111→      setLoadingBuildings(false);
   112→      return;
   113→    }
   114→    setZoomHint(false);
   115→    const b = map.getBounds();
   116→    const qs = new URLSearchParams({
   117→      west: String(b.getWest().toFixed(5)),
   118→      south: String(b.getSouth().toFixed(5)),
   119→      east: String(b.getEast().toFixed(5)),
   120→      north: String(b.getNorth().toFixed(5)),
   121→    });
   122→    setLoadingBuildings(true);
   123→    setBuildingsError(null);
   124→    try {
   125→      const res = await fetch(`/api/buildings?${qs}`);
   126→      if (!res.ok) {
   127→        const d = await res.json().catch(() => ({}));
   128→        if (d.error === "BBOX_TOO_BIG") {
   129→          layerRef.current.clearLayers();
   130→          setBuildingsError("zoom");
   131→        } else {
   132→          setBuildingsError("service");
   133→        }
   134→        return;
   135→      }
   136→      const data = await res.json();
   137→      drawBuildings(data.buildings || []);
   138→    } catch {
   139→      setBuildingsError("service");
   140→    } finally {
   141→      setLoadingBuildings(false);
   142→    }
   143→     
   144→  }, []);
   145→
   146→  // recarrega edificis quan canvia l'idioma/sessió per refrescar colors i accés
   147→  useEffect(() => {
   148→    loadBuildingsInView();
   149→     
   150→  }, [locale]);
   151→
   152→  const drawBuildings = useCallback((buildings: any[]) => {
   153→    const L = LRef.current;
   154→    const layer = layerRef.current;
   155→    if (!L || !layer) return;
   156→    layer.clearLayers();
   157→    for (const b of buildings) {
   158→      let color = "#9aa5a0";
   159→      if (b.munRent) color = rentColor(b.munRent);
   160→      const poly = L.polygon(b.ring.map((p: number[]) => [p[1], p[0]]), {
   161→        color: "#3d4a42",
   162→        weight: 0.7,
   163→        fillColor: color,
   164→        fillOpacity: 0.55,
   165→      });
   166→      poly.on("click", (e: any) => {
   167→        L.DomEvent.stopPropagation(e);
   168→        onSelectBuilding(b, e.latlng.lat, e.latlng.lng, poly);
   169→      });
   170→      poly.addTo(layer);
   171→    }
   172→     
   173→  }, []);
   174→
   175→  const lookupSeqRef = useRef(0);
   176→
   177→  const onSelectBuilding = useCallback(async (b: any, lat: number, lng: number, poly: any) => {
   178→    const L = LRef.current;
   179→    const map = mapRef.current;
   180→    if (L && map) {
   181→      if (selectedRef.current) {
   182→        selectedRef.current.setStyle({ color: "#3d4a42", weight: 0.7, fillOpacity: 0.55 });
   183→      }
   184→      poly.setStyle({ color: "#e8673c", weight: 3, fillOpacity: 0.8 });
   185→      selectedRef.current = poly;
   186→    }
   187→    // invalida respuestas de consultas anteriores (evita carreras fuera de orden)
   188→    const seq = ++lookupSeqRef.current;
   189→    setPanelState("loading");
   190→    setLookup(null);
   191→    setDemoNotice(null);
   192→    try {
   193→      const res = await fetch("/api/lookup", {
   194→        method: "POST",
   195→        headers: { "Content-Type": "application/json" },
   196→        body: JSON.stringify({ lat, lng, buildingId: b.id }),
   197→      });
   198→      if (seq !== lookupSeqRef.current) return; // respuesta obsoleta
   199→      if (res.status === 402) {
   200→        const d = await res.json().catch(() => ({}));
   201→        setLookup(d);
   202→        setPanelState("paywall");
   203→        return;
   204→      }
   205→      const d: LookupResult = await res.json();
   206→      if (seq !== lookupSeqRef.current) return; // respuesta obsoleta
   207→      setLookup(d);
   208→      setPanelState(d.kind === "full" ? "full" : "teaser");
   209→      if (d.kind === "full" && d.quota) setQuota(d.quota);

   211→      if (seq === lookupSeqRef.current) setPanelState("closed");
   212→    }

   214→
   215→  // ---- Cercador de municipis ----
   216→  useEffect(() => {

   218→      setResults([]);
   219→      return;
   220→    }
   221→    const id = setTimeout(async () => {
   222→      try {
   223→        const res = await fetch(`/api/municipalities?q=${encodeURIComponent(query.trim())}&limit=8`);
   224→        const d = await res.json();
   225→        setResults(d.municipalities || []);
   226→        setShowResults(true);
   227→      } catch {}
   228→    }, 220);
   229→    return () => clearTimeout(id);
   230→  }, [query]);
   231→
   232→  const selectMunicipality = useCallback(async (mun: MunLite) => {
   233→    setShowResults(false);
   234→    setQuery(mun.name);
   235→    setSelectedMun(mun);
   236→    const map = mapRef.current;
   237→    if (!map) return;
   238→    // límit municipal real (ICGC amb cau) + fit
   239→    try {
   240→      const res = await fetch(`/api/municipalities/${mun.code}`);
   241→      const d = await res.json();
   242→      if (d.boundary) {
   243→        const L = LRef.current;
   244→        boundaryRef.current?.clearLayers();
   245→        const layer = L.geoJSON(d.boundary, {
   246→          style: { color: "#0f2e21", weight: 2, fill: false, dashArray: "4 3" },
   247→        });
   248→        layer.addTo(boundaryRef.current);
   249→        map.fitBounds(layer.getBounds().pad(0.05));
   250→        // si ja tenim prou zoom, carreguem edificis immediatament
   251→        if (map.getZoom() >= 16) loadBuildingsInView();
   252→      } else {
   253→        map.setView([mun.lat, mun.lng], 16);
   254→        setTimeout(() => loadBuildingsInView(), 400);
   255→      }
   256→    } catch {
   257→      map.setView([mun.lat, mun.lng], 16);
   258→    }
   259→  }, [loadBuildingsInView]);
   260→
   261→  const doCheckout = async (plan: "PREMIUM" | "PRO") => {
   262→    try {
   263→      const res = await fetch("/api/stripe/checkout", {
   264→        method: "POST",
   265→        headers: { "Content-Type": "application/json" },
   266→        body: JSON.stringify({ plan }),
   267→      });
   268→      const d = await res.json();
   269→      if (d.ok && d.mode === "STRIPE" && d.url) {
   270→        window.location.href = d.url;
   271→      } else if (d.ok && d.mode === "DEMO") {
   272→        setDemoNotice(d.message || "Demo");
   273→        setPanelState("full");
   274→        await auth.refresh();
   275→      } else {
   276→        setDemoNotice(d.message || d.error || "Error");
   277→      }
   278→    } catch {
   279→      setDemoNotice("Error de xarxa");
   280→    }
   281→  };
   282→
   283→  return (
   284→    <div className="flex-1 flex flex-col md:flex-row relative md:h-[calc(100vh-4rem)] md:overflow-hidden">
   285→      {/* MAPA */}
   286→      <div className="relative flex-1 min-h-[55vh] md:h-full md:min-h-[520px] md:flex-none md:basis-auto md:grow">
   287→        <div ref={mapDivRef} className="absolute inset-0 z-0" />
   288→
   289→        {/* Cercador */}
   290→        <div className="absolute top-3 left-3 right-3 z-[500] max-w-md">
   291→          <div className="relative">
   292→            <div className="bg-white rounded-xl shadow-lg border border-[#e5e1da] flex items-center px-3">
   293→              <Search className="w-4 h-4 text-[#6b6f68] shrink-0" />
   294→              <input
   295→                value={query}
   296→                onChange={(e) => setQuery(e.target.value)}
   297→                onFocus={() => results.length && setShowResults(true)}
   298→                onBlur={() => setTimeout(() => setShowResults(false), 180)}
   299→                placeholder={t.map.searchPlaceholder}
   300→                className="w-full px-2 py-3 text-sm outline-none bg-transparent"
   301→                aria-label={t.map.searchPlaceholder}
   302→              />
   303→              {query && (
   304→                <button onClick={() => { setQuery(""); setSelectedMun(null); boundaryRef.current?.clearLayers(); }}>
   305→                  <X className="w-4 h-4 text-[#6b6f68]" />
   306→                </button>
   307→              )}
   308→            </div>
   309→            {showResults && results.length > 0 && (
   310→              <div className="absolute mt-1 w-full bg-white rounded-xl shadow-xl border border-[#e5e1da] overflow-hidden max-h-80 overflow-y-auto pr-scroll">
   311→                {results.map((m) => (
   312→                  <button
   313→                    key={m.code}
   314→                    onMouseDown={(e) => e.preventDefault()}
   315→                    onClick={() => selectMunicipality(m)}
   316→                    className="w-full text-left px-4 py-2.5 hover:bg-[#f0ede8] flex items-center justify-between gap-2"
   317→                  >
   318→                    <span className="text-sm font-medium">{m.name}</span>
   319→                    <span className="text-xs text-[#6b6f68] flex items-center gap-1.5">
   320→                      {m.rentLatest ? (
   321→                        <span className="font-bold text-[#0f2e21]">{Math.round(m.rentLatest)} €/mes</span>
   322→                      ) : (
   323→                        <span>—</span>
   324→                      )}
   325→                      {m.isTense && <Flame className="w-3.5 h-3.5 text-[#e8673c]" />}
   326→                    </span>
   327→                  </button>
   328→                ))}
   329→              </div>
   330→            )}
   331→          </div>
   332→        </div>
   333→
   334→        {demoNotice && (
   335→          <div className="absolute top-16 left-3 right-3 z-[500] max-w-md bg-[#0f2e21] text-white text-xs rounded-xl p-3 shadow-lg flex justify-between gap-2">
   336→            <span>{demoNotice}</span>
   337→            <button onClick={() => setDemoNotice(null)}><X className="w-3.5 h-3.5" /></button>
   338→          </div>
   339→        )}
   340→
   341→        {/* Estats */}
   342→        <div className="absolute bottom-3 left-3 z-[500] flex flex-col gap-2 max-w-[85%]">
   343→          {(zoomHint || buildingsError === "zoom") && (
   344→            <div className="bg-white/95 rounded-lg px-3 py-2 text-xs shadow border border-[#e5e1da] flex items-center gap-2">
   345→              <Info className="w-3.5 h-3.5 text-[#e8673c]" /> {t.map.zoomHint}
   346→            </div>
   347→          )}
   348→          {loadingBuildings && (
   349→            <div className="bg-white/95 rounded-lg px-3 py-2 text-xs shadow border border-[#e5e1da] flex items-center gap-2">
   350→              <span className="w-3 h-3 rounded-full border-2 border-[#e8673c] border-t-transparent animate-spin" />
   351→              {t.map.loadingBuildings}
   352→            </div>
   353→          )}
   354→          {buildingsError === "service" && (
   355→            <div className="bg-[#fdeae2] rounded-lg px-3 py-2 text-xs shadow border border-[#f2c4ac] text-[#b0431c]">
   356→              {locale === "ca"
   357→                ? "El servei de cartografia no està disponible ara mateix. Torna-ho a provar."
   358→                : "El servicio de cartografía no está disponible ahora mismo. Inténtalo de nuevo."}
   359→            </div>
   360→          )}
   361→        </div>
   362→
   363→        {/* Llegenda */}
   364→        <div className="absolute bottom-3 right-3 z-[500] bg-white/95 rounded-lg px-3 py-2 text-[10px] shadow border border-[#e5e1da] hidden sm:block">
   365→          <div className="font-semibold mb-1">{t.map.legend.low} → {t.map.legend.high}</div>
   366→          <div className="flex gap-1 items-center">
   367→            {["#63b57c", "#93c47d", "#e5c46a", "#e89a4b", "#dd5a35"].map((c) => (
   368→              <span key={c} className="w-5 h-3 rounded-sm inline-block" style={{ background: c }} />
   369→            ))}
   370→          </div>
   371→          <div className="mt-1 flex items-center gap-1 text-[#6b6f68]">
   372→            <Flame className="w-3 h-3 text-[#e8673c]" /> {t.map.legend.tense}
   373→          </div>
   374→        </div>
   375→      </div>
   376→
   377→      {/* PANELL DETALL — desktop lateral, mobile fulla inferior */}
   378→      {panelState !== "closed" && (
   379→        <div className="md:w-[400px] md:h-full w-full max-h-[70vh] md:max-h-full bg-white border-t md:border-t-0 md:border-l border-[#e5e1da] shadow-2xl z-[600] overflow-y-auto pr-scroll shrink-0 fixed bottom-0 left-0 right-0 md:static">
   380→          <div className="p-4 md:p-5">
   381→            <div className="flex items-start justify-between gap-2 mb-3">
   382→              <h2 className="font-extrabold text-lg leading-tight">
   383→                {panelState === "loading" && t.common.loading}
   384→                {lookup?.municipality?.name || "—"}
   385→                {lookup?.municipality?.comarca && (
   386→                  <span className="block text-xs font-normal text-[#6b6f68]">{lookup.municipality.comarca}</span>
   387→                )}
   388→              </h2>
   389→              <button onClick={() => { setPanelState("closed"); if (selectedRef.current) { selectedRef.current.setStyle({ color: "#3d4a42", weight: 0.7, fillOpacity: 0.55 }); selectedRef.current = null; } }} className="p-1.5 rounded-lg hover:bg-[#f0ede8]">
   390→                <X className="w-5 h-5" />
   391→              </button>
   392→            </div>
   393→
   394→            {panelState === "loading" && (
   395→              <div className="py-10 flex justify-center">
   396→                <span className="w-8 h-8 rounded-full border-4 border-[#e8673c] border-t-transparent animate-spin" />
   397→              </div>
   398→            )}
   399→
   400→            {/* TEASER (sense sessió) */}
   401→            {panelState === "teaser" && (
   402→              <div className="space-y-4">
   403→                {lookup?.municipality?.isTense && (
   404→                  <TenseBadge note={t.tense.badgeEsNote} />
   405→                )}
   406→                <div className="rounded-xl border border-[#e5e1da] bg-[#faf9f7] p-4">
   407→                  <div className="text-xs font-semibold text-[#6b6f68] uppercase tracking-wide">{t.map.panel.avgRent}</div>
   408→                  <div className="mt-2 select-none blur-[7px] text-4xl font-black text-[#0f2e21]">1.134 €</div>
   409→                  <div className="mt-1 text-xs text-[#6b6f68]">Incasòl · {lookup?.teaser?.period ?? ""}</div>
   410→                </div>
   411→                <div className="rounded-xl bg-[#0f2e21] text-white p-4">
   412→                  <div className="font-bold">{t.map.teaser.title}</div>
   413→                  <p className="text-sm mt-1 opacity-90 leading-relaxed">{t.map.teaser.desc}</p>
   414→                  <div className="mt-3 flex flex-col gap-2">
   415→                    <button onClick={() => navigate("/registre")} className="w-full py-2.5 rounded-lg bg-[#e8673c] font-bold text-sm hover:bg-[#d95a31]">
   416→                      {t.map.teaser.cta}
   417→                    </button>
   418→                    <button onClick={() => navigate("/entrada")} className="w-full py-2 rounded-lg border border-white/30 text-sm hover:bg-white/10">
   419→                      {t.map.teaser.login}
   420→                    </button>
   421→                  </div>
   422→                </div>
   423→              </div>
   424→            )}
   425→
   426→            {/* PAYWALL (quota exhaurida) */}
   427→            {panelState === "paywall" && (
   428→              <div className="space-y-4">
   429→                <div className="rounded-xl bg-[#fdeae2] border border-[#f2c4ac] p-4 text-center">
   430→                  <Building2 className="w-8 h-8 text-[#b0431c] mx-auto" />
   431→                  <div className="font-bold mt-2 text-[#b0431c]">{t.map.paywall.title}</div>
   432→                  <p className="text-sm mt-1 text-[#4b5349]">{t.map.paywall.desc}</p>
   433→                </div>
   434→                <button onClick={() => navigate("/preus")} className="w-full py-3 rounded-xl bg-[#e8673c] text-white font-bold hover:bg-[#d95a31]">
   435→                  {t.map.paywall.cta}
   436→                </button>
   437→              </div>
   438→            )}
   439→
   440→            {/* FULL */}
   441→            {panelState === "full" && lookup?.municipality && (
   442→              <FullPanel
   443→                lookup={lookup}
   444→                quota={quota}
   445→                t={t}
   446→                navigate={navigate}
   447→                doCheckout={doCheckout}
   448→              />
   449→            )}
   450→          </div>
   451→        </div>
   452→      )}
   453→    </div>
   454→  );
   455→}

   457→function TenseBadge({ note }: { note: string }) {
   458→  const { t } = useApp();
   459→  return (
   460→    <div className="rounded-xl bg-[#fdeae2] border border-[#f2c4ac] px-3 py-2.5 flex items-start gap-2">
   461→      <Flame className="w-4 h-4 text-[#b0431c] mt-0.5 shrink-0" />
   462→      <div>
   463→        <div className="text-sm font-bold text-[#b0431c]">{t.tense.badgeFull}</div>

   465→      </div>
   466→    </div>
   467→  );
   468→}
   469→
   470→function FullPanel({
   471→  lookup, quota, t, navigate, doCheckout,
   472→}: {
   473→  lookup: LookupResult;
   474→  quota: { used: number; limit: number } | null;
   475→  t: any;
   476→  navigate: (r: string) => void;
   477→  doCheckout: (p: "PREMIUM" | "PRO") => void;
   478→}) {
   479→  const { locale, auth } = useApp();
   480→  const mun = lookup.municipality!;
   481→  const [yourRent, setYourRent] = useState("");
   482→  const [yourM2, setYourM2] = useState("");
   483→  const [reportOpen, setReportOpen] = useState(false);
   484→  const [reportSent, setReportSent] = useState(false);
   485→
   486→  const hist = mun.rentHistory ? Object.entries(mun.rentHistory).sort() : [];
   487→  const histShow = hist.slice(-8);
   488→  const maxVal = Math.max(...histShow.map(([, v]) => Number(v)), 1);
   489→
   490→  const trend =
   491→    mun.rentLatest && mun.rentPrevYear
   492→      ? ((mun.rentLatest - mun.rentPrevYear) / mun.rentPrevYear) * 100
   493→      : null;
   494→
   495→  const yourRentNum = parseFloat(yourRent);
   496→  const delta =
   497→    isFinite(yourRentNum) && mun.rentLatest
   498→      ? ((yourRentNum - mun.rentLatest) / mun.rentLatest) * 100
   499→      : null;
   500→
   501→  const submitReport = async () => {
   502→    await fetch("/api/reports", {
   503→      method: "POST",
   504→      headers: { "Content-Type": "application/json" },
   505→      body: JSON.stringify({
   506→        municipalityCode: mun.code,
   507→        buildingId: lookup.building?.id || null,
   508→        offerEurMonth: yourRentNum || undefined,
   509→        surfaceM2: parseFloat(yourM2) || undefined,
   510→      }),
   511→    });
   512→    setReportSent(true);
   513→    setReportOpen(false);
   514→  };
   515→
   516→  return (
   517→    <div className="space-y-4">
   518→      {mun.isTense && <TenseBadge note={t.tense.badgeEsNote} />}
   519→
   520→      {/* PREU MITJÀ REAL */}
   521→      <div className="rounded-xl border border-[#e5e1da] bg-[#faf9f7] p-4">
   522→        <div className="text-xs font-semibold text-[#6b6f68] uppercase tracking-wide">{t.map.panel.avgRent}</div>
   523→        <div className="mt-1 flex items-end gap-2">
   524→          <span className="text-4xl font-black text-[#0f2e21]">
   525→            {mun.rentLatest ? `${mun.rentLatest.toFixed(0)} €` : "—"}
   526→          </span>
   527→          <span className="text-sm text-[#6b6f68] mb-1">{t.map.panel.perMonth}</span>
   528→          {trend !== null && (
   529→            <span className={`mb-1.5 ml-auto text-xs font-bold px-2 py-1 rounded-md flex items-center gap-1 ${
   530→              trend > 0 ? "bg-[#fdeae2] text-[#b0431c]" : trend < 0 ? "bg-[#e6f4ea] text-[#1e7a41]" : "bg-[#f0ede8]"
   531→            }`}>
   532→              {trend > 0 ? <TrendingUp className="w-3 h-3" /> : trend < 0 ? <TrendingDown className="w-3 h-3" /> : <Minus className="w-3 h-3" />}
   533→              {trend > 0 ? "+" : ""}{trend.toFixed(1)}%
   534→            </span>
   535→          )}
   536→        </div>
   537→        <div className="mt-2 text-xs text-[#6b6f68] space-y-0.5">
   538→          <div>{t.map.panel.period}: <b>{mun.rentPeriod || "—"}</b></div>
   539→          {mun.contractsLatest !== null && (
   540→            <div>{t.map.panel.contracts}: <b>{mun.contractsLatest.toLocaleString(locale === "ca" ? "ca-ES" : "es-ES")}</b></div>
   541→          )}
   542→          <div className="flex items-center gap-1">
   543→            {t.map.panel.source}
   544→            <ExternalLink className="w-3 h-3" />
   545→          </div>
   546→        </div>
   547→      </div>
   548→
   549→      {/* EDIFICI */}
   550→      {(lookup.building?.yearBuilt || lookup.building?.refcat) && (
   551→        <div className="rounded-xl border border-[#e5e1da] p-4">
   552→          <div className="text-xs font-semibold text-[#6b6f68] uppercase tracking-wide mb-2">
   553→            {locale === "ca" ? "Edifici" : "Edificio"}
   554→          </div>
   555→          <div className="flex justify-between text-sm py-1">
   556→            <span className="text-[#6b6f68]">{t.map.panel.yearBuilt}</span>
   557→            <b>
   558→              {lookup.building?.yearBuilt
   559→                ? `${lookup.building.yearBuilt.year} (${lookup.building.yearBuilt.source === "CADASTRE_DNP" ? "Cadastre" : lookup.building.yearBuilt.source})`
   560→                : lookup.building?.refcat ? t.map.panel.yearUnavailable : "—"}
   561→            </b>
   562→          </div>
   563→          {lookup.building?.refcat && (
   564→            <div className="flex justify-between text-sm py-1">
   565→              <span className="text-[#6b6f68]">{t.map.panel.refcat}</span>
   566→              <b className="font-mono text-xs">{lookup.building.refcat}</b>
   567→            </div>
   568→          )}
   569→        </div>
   570→      )}
   571→
   572→      {/* COMPARADOR */}
   573→      <div className="rounded-xl border-2 border-[#0f2e21] p-4">
   574→        <div className="font-bold text-[#0f2e21]">{t.map.panel.compare}</div>
   575→        <div className="mt-3 grid grid-cols-2 gap-2">
   576→          <div>
   577→            <label className="text-xs text-[#6b6f68]">{t.map.panel.yourRent}</label>
   578→            <input
   579→              value={yourRent}
   580→              onChange={(e) => setYourRent(e.target.value.replace(/[^0-9.]/g, ""))}
   581→              inputMode="decimal"
   582→              className="w-full mt-1 px-3 py-2 rounded-lg border border-[#e5e1da] text-sm outline-none focus:border-[#e8673c]"
   583→              placeholder="950"
   584→            />
   585→          </div>
   586→          <div>
   587→            <label className="text-xs text-[#6b6f68]">{t.map.panel.yourSurface}</label>
   588→            <input
   589→              value={yourM2}
   590→              onChange={(e) => setYourM2(e.target.value.replace(/[^0-9.]/g, ""))}
   591→              inputMode="decimal"
   592→              className="w-full mt-1 px-3 py-2 rounded-lg border border-[#e5e1da] text-sm outline-none focus:border-[#e8673c]"
   593→              placeholder="65"
   594→            />
   595→          </div>
   596→        </div>
   597→        {delta !== null && (
   598→          <div className={`mt-3 rounded-lg p-3 text-center font-bold ${
   599→            delta > 5 ? "bg-[#fdeae2] text-[#b0431c]" : delta < -5 ? "bg-[#e6f4ea] text-[#1e7a41]" : "bg-[#f0ede8] text-[#4b5349]"
   600→          }`}>
   601→            {delta > 0 ? "+" : ""}{delta.toFixed(1)}%{" "}
   602→            <span className="font-normal text-sm">
   603→              {delta > 5 ? t.map.panel.above : delta < -5 ? t.map.panel.below : t.map.panel.atMarket}
   604→            </span>
   605→          </div>
   606→        )}
   607→        <button
   608→          onClick={() => setReportOpen(true)}
   609→          className="mt-3 w-full py-2 rounded-lg border border-[#e5e1da] text-xs font-semibold hover:bg-[#f0ede8] flex items-center justify-center gap-1.5"
   610→        >
   611→          <MapPin className="w-3.5 h-3.5" />
   612→          {reportSent ? t.map.panel.reportThanks : t.map.panel.report}
   613→        </button>
   614→        {reportOpen && !reportSent && (
   615→          <div className="mt-2 flex gap-2">
   616→            <input
   617→              autoFocus
   618→              value={yourRent}
   619→              onChange={(e) => setYourRent(e.target.value.replace(/[^0-9.]/g, ""))}
   620→              className="flex-1 px-3 py-2 rounded-lg border border-[#e5e1da] text-sm"
   621→              placeholder={t.map.panel.yourRent}
   622→            />
   623→            <button onClick={submitReport} className="px-4 py-2 rounded-lg bg-[#0f2e21] text-white text-sm font-semibold">
   624→              {t.map.panel.reportBtn}
   625→            </button>
   626→          </div>
   627→        )}
   628→      </div>
   629→
   630→      {/* HISTÒRIC */}
   631→      {histShow.length > 1 && (
   632→        <div className="rounded-xl border border-[#e5e1da] p-4">
   633→          <div className="text-xs font-semibold text-[#6b6f68] uppercase tracking-wide mb-3">{t.map.panel.history}</div>
   634→          <div className="flex items-end gap-1.5 h-24">
   635→            {histShow.map(([year, v]) => (
   636→              <div key={year} className="flex-1 flex flex-col items-center gap-1">
   637→                <div
   638→                  className="w-full rounded-t bg-[#0f2e21]/85 min-h-[3px]"
   639→                  style={{ height: `${(Number(v) / maxVal) * 100}%` }}
   640→                  title={`${year}: ${v} €`}
   641→                />
   642→                <span className="text-[9px] text-[#6b6f68]">{year.slice(2)}</span>
   643→              </div>
   644→            ))}
   645→          </div>
   646→          <div className="text-[10px] text-[#6b6f68] mt-2">{t.map.panel.source}</div>
   647→        </div>
   648→      )}
   649→
   650→      {/* QUOTA + UPGRADE */}
   651→      {auth.user?.plan === "FREE" && quota && (
   652→        <div className="rounded-xl bg-[#f0ede8] p-4 flex items-center justify-between gap-3">
   653→          <div className="text-sm">
   654→            <b>{quota.limit - quota.used}</b> {t.map.panel.quotaLeft}
   655→          </div>
   656→          <button onClick={() => navigate("/preus")} className="px-3 py-2 rounded-lg bg-[#e8673c] text-white text-sm font-bold hover:bg-[#d95a31] whitespace-nowrap">
   657→            {t.map.panel.upgrade}
   658→          </button>
   659→        </div>
   660→      )}
   661→
   662→      {auth.user && auth.user.plan !== "FREE" && (
   663→        <button onClick={() => doCheckout(auth.user!.plan === "FREE" ? "PREMIUM" : "PRO")} className="hidden" />
   664→      )}
   665→    </div>
   666→  );
   667→}
